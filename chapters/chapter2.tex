\khmertopic{១. សមីការបម្លាស់ទីភាគល្អិត ({
	\englishfont
	Equation of Particle Displacement})}

\textbf{ក. និយមន័យ៖}រលកគឺជាការបញ្ជូនរញ្ជួយ
ឬលំយោលបន្តបន្ទាប់គ្នាក្នុងមជ្ឈដ្ឋានតែមួយ។
តាមច្បាប់ទី ២ ញូតុន និងកម្លាំងយឺត ($F=-kx$
) យើងទទួលបានសមីការឌីផេរ៉ង់ស្យែល
$\frac{d^2x}{dt^2} + \omega^2x = 0$
ដែលមានចម្លើយជាអនុគមន៍ស៊ីនុយសូអ៊ីត៖
\begin{equation}
	x = A \sin(\omega t + \varphi) \quad \text{ឬ}
	\quad x = A \cos(\omega t + \varphi)
\end{equation}

\khmertopic{២. ល្បឿន និងសំទុះនៃភាគល្អិតរញ្ជួយ}
\begin{itemize}
	\item \textbf{ល្បឿន ($v$):}$
	v = \frac{dx}{dt} = \omega A
	\cos(\omega t + \varphi)$។ ល្បឿនអតិបរមា
	$v_{max} = \omega A$ (នៅទីតាំងលំនឹង)។
	\item \textbf{សំទុះ ($a$):}$
	a = \frac{dv}{dt} = -\omega^2 A
	\sin(\omega t + \varphi) = -\omega^2 x$
	។ សំទុះអតិបរមា $a_{max} = \omega^2 A$
	(នៅអំព្លីទុត)។
\end{itemize}

\khmertopic
{៤. វិធីសាស្ត្រក្នុងការធ្វើសំណង់ប្រេណែល (
		{\englishfont Fresnel Construction})}

វិធីសាស្ត្រប្រណែល
ប្រើសម្រាប់បូកបញ្ចូលរលកពីរដែលមានប្រេកង់ដូចគ្នា
តែមានអំព្លីទុត និងផាសដើមខុសគ្នា
ដោយប្រើការបូកវិចទ័រ។

\textbf{ជំហានអនុវត្ត (ឧទាហរណ៍៖ បូករលក $x_1$ ផាស
	$\varphi_1=30^\circ$ និង $x_2$ ផាស
	$\varphi_2=90^\circ$):}
\begin{center}
	\begin{tikzpicture}[>=Stealth, scale=0.8]
		\draw[->, thick, dashed] (0,0)
		-- (4,0) node[right]
			{អ័ក្សគល់ (Reference)};
		\coordinate (O) at (0,0);
		\coordinate (A1) at (30:3);
		\coordinate (A2) at ($(A1) + (90:4)$);
		% Calculate tip of A2
		
		\draw[->, thick, khmerred] (O)

			{$\vec{{\englishfont A}}_1$};
		\draw[->, thick, blue] (A1)

			{$\vec{{\englishfont A}}_2$};
		
		% Resultant Vector
		\draw[->, ultra thick, purple] (O)
		-- (A2) node[midway, above left]{$
		\vec{{\englishfont A}} \text{ (សរុប)}$};
		
		% Angles
		\draw (1,0)

		above
		right] {$\varphi_2$};
		\draw (1.5,0)
		arc (0:56:1.5) node[midway, right=0.1cm]
			{$\varphi$};
	\end{tikzpicture}
\end{center}

\textbf{រូបមន្តគណនាទំហំ និងមុំវិចទ័រសរុប (
	តាមទ្រឹស្តីបទកូស៊ីនុស):}
\begin{align}
	A &=\sqrt{A_1^2 + A_2^2 + 2A_1A_2
	\cos(\varphi_2 - \varphi_1)} \\
	\tan \varphi &=
	\frac{A_1\sin\varphi_1 + A_2\sin\varphi_2}
	{A_1\cos\varphi_1 + A_2\cos\varphi_2}
\end{align}

\noindent \textbf{\color{brownheader}ឧទាហរណ៍៖ }
គេឲ្យរលក២ដូចខាងក្រោម៖
\[
	y_1=(5 \mathrm{cm}) \sin (10 \pi t + \pi)\]
\[
	y_2=(5 \mathrm{cm})
	\sin (10 \pi t + \frac{\pi}{2})
\]
\noindent សរសេរសមីការលំយោលតម្រួតនៃរលកទាំងពីរ។\\
\noindent \textbf{\color{brownheader}ចម្លើយ៖}\\

\noindent\textbf
{ជំហានទី១៖ ដកស្រង់ទិន្នន័យពីសមីការរលកទាំងពីរ}\\
ដោយប្រៀបធៀបនឹងទម្រង់ទូទៅ
$y = A\sin(\omega t + \varphi)$ យើងបាន៖
\[
	A_1 = 5\,\mathrm{cm}, \quad \varphi_1 = \pi
	\qquad ; \qquad
	A_2 = 5\,\mathrm{cm}, \quad \varphi_2 =
	\frac{\pi}{2}
\]
រលកទាំងពីរមានប៉ូល្សាស្យុងដូចគ្នា
$\omega = 10\pi\ \mathrm{rad/s}$
ដូច្នេះយើងអាចប្រើសំណង់ប្រេណែលបាន។
ផលសងផាស៖
\[
	\Delta\varphi = \varphi_2 - \varphi_1 =
	\frac{\pi}{2} - \pi = -\frac{\pi}{2}
\]

\noindent
\textbf{ជំហានទី២៖ គូសសំណង់ប្រេណែល (បូកវិចទ័រ)}\\
តាង $\vec{A}_1$ ធ្វើមុំ $\varphi_1 = \pi$ (
$180^\circ$) និង $\vec{A}_2$ ធ្វើមុំ
$\varphi_2 = \frac{\pi}{2}$ ($90^\circ$
) ជាមួយអ័ក្សគល់៖
\begin{center}
	\begin{tikzpicture}[>=Stealth, scale=0.8]
		\draw[->, thick, dashed] (-4,0)
		-- (2,0) node[right] {អ័ក្សគល់};
		\draw[->, thick, dashed] (0,-0.6)
		-- (0,3.6);
		\coordinate (O) at (0,0);
		\coordinate (A1) at (180:3);
		\coordinate (A2) at ($(A1) + (90:3)$);
		
		\draw[->, thick, khmerred] (O)
		-- (A1) node[midway, below=2pt]
			{$\vec{{\englishfont A}}_1 = 5$};
		\draw[->, thick, blue] (A1)
		-- (A2) node[midway, left=2pt]
			{$\vec{{\englishfont A}}_2 = 5$};
		\draw[->, ultra thick, purple] (O)
		-- (A2)
		node[midway, above left=-1pt]
			{
			$\vec{{\englishfont A}} = 5\sqrt{2}$}
		;
		
		\draw[thin] (1.0,0) arc (0:180:1.0);
		\node at (-1.95,0.45) {$\varphi_1 = \pi$};
		\draw[thin] (0.55,0) arc (0:135:0.55);
		\node at (0.72,0.68) {$\varphi$};
		\filldraw[black] (O)
		circle (1.5pt) node[below right]
			{${\englishfont O}$};
	\end{tikzpicture}
\end{center}
កត់សម្គាល់៖ ដោយ $\Delta\varphi = -\dfrac{\pi}{2}$
វិចទ័រទាំងពីរកែងគ្នា
ដូច្នេះត្រីកោណបូកវិចទ័រជាត្រីកោណកែងសមបាត។

\noindent \textbf{ជំហានទី៣៖ គណនាអំព្លីទុតសរុប $A$}
\\
តាមទ្រឹស្តីបទកូស៊ីនុស៖
\begin{align*}
	A &=\sqrt{A_1^2 + A_2^2 + 2A_1A_2
	\cos(\varphi_2 - \varphi_1)} \\
	&=\sqrt{5^2 + 5^2 + 2(5)(5)
		\cos\left(-\frac{\pi}{2}\right)} \\
	&= \sqrt{25 + 25 + 50 \times 0} =
	\sqrt{50} \\
	A &= 5\sqrt{2} \approx 7{,}07\ \mathrm{cm}
\end{align*}

\noindent
\textbf{ជំហានទី៤៖ គណនាផាសដើមសរុប $\varphi$}\\
\begin{align*}
	\tan\varphi &=
	\frac{A_1\sin\varphi_1 + A_2\sin\varphi_2}
	{A_1\cos\varphi_1 + A_2\cos\varphi_2}
	=\frac{5\sin\pi + 5\sin\frac{\pi}{2}}
	{5\cos\pi + 5\cos\frac{\pi}{2}} \\
	&= \frac{5(0) + 5(1)}{5(-1) + 5(0)} =
	\frac{5}{-5} = -1
\end{align*}
ដោយភាគយក $> 0$ ($\sin\varphi > 0$) និងភាគបែង $< 0$
($\cos\varphi < 0$) នោះ $\varphi$
ស្ថិតក្នុងកាដ្រង់ទី២៖
\[
	\varphi = \pi - \frac{\pi}{4} = \frac{3\pi}{4}
\]

\noindent
\textbf{ជំហានទី៥៖ សរសេរសមីការលំយោលតម្រួត}\\
ដោយ $\omega = 10\pi\ \mathrm{rad/s}$
មិនប្រែប្រួល យើងបាន៖
\begin{equation*}
	\boxed{\,y = y_1 + y_2 =
		\left(5\sqrt{2}\ \mathrm{cm}\right)\sin
		\left(10\pi t + \frac{3\pi}{4}\right)\,}
\end{equation*}

\noindent \textbf{ជំហានទី៦៖ ផ្ទៀងផ្ទាត់លទ្ធផល}\\
ជំនួស $t = 0$ ក្នុងសមីការដើម៖
$y_1(0) = 5\sin\pi = 0$ និង
$y_2(0) = 5\sin\frac{\pi}{2} = 5$ ដូច្នេះ
$y(0) = 5\ \mathrm{cm}$។\\
ជំនួស $t = 0$ ក្នុងលទ្ធផល៖$
y(0) = 5\sqrt{2}\sin\frac{3\pi}{4} = 5\sqrt{2}
\times \frac{\sqrt{2}}{2} = 5\ \mathrm{cm}$
។ ត្រូវ។

ដែលមានអំព្លីទុត ប្រវែងរលក និងប្រេកង់ដូចគ្នា
ប៉ុន្តែធ្វើដំណើរក្នុងទិសដៅផ្ទុយគ្នា។

\textbf{ខ. សមីការរលកជញ្ច្រុំ៖}
តាងរលកចូល៖ $y_1 = a \sin(kx - \omega t)$
និង រលកចំណាំងផ្លាត៖ $y_2 = a \sin(kx + \omega t)$។
តាមគោលការណ៍តម្រួត និងរូបមន្តត្រីកោណមាត្រ
យើងទទួលបានសមីការ៖
\begin{equation}
	y(x,t) = [2a \sin(kx)] \cos(\omega t) \quad
	\text{ឬ} \quad y(x,t) = A_{SW} \cos(\omega t)
\end{equation}

\textbf{គ. ទីតាំងថ្នាំង និងទីតាំងពោះ (
		{\englishfont Nodes and Antinodes}):}\\
\begin{itemize}
	[label=\faChevronRight, noitemsep, topsep=
	2pt,
		parsep=0pt]
	\item \textbf{ទីតាំងពោះ}
	គឺជាទីតាំងទាំងឡាយណាដែលអំព្លីទុតរលកជញ្ចុំមានតម្លៃអតិបរមា។
	គេសរសេរវាដោយ $2a \sin kx$ កើនដល់អតិបរមា។
	\item \textbf{ទីតាំងថ្នាំង}
	គឺជាទីតាំងទាំងឡាយណាដែលអំព្លីទុតរលកជញ្ចុំមានតម្លៃស្មើសូន្យ។
	គេសរសេរវាដោយ $2a \sin kx = 0$។
\end{itemize}

	$L = n\frac{\lambda}{2}$ ដែល $n$ ជាចំនួនពិត។
\end{itemize}

\textbf{\textcolor{bacpink}{និយមន័យ៖}}
អាំងទែផេរ៉ង់សូរជារលកសូរដែលកើតឡើងចេញពីប្រភព២ដែលមានអំព្លីទុតរលកផ្គូបមានតម្លៃអតិបរមាហៅថា \textbf{\textcolor{bacpink}{ទីតាំងសូរអតិបរមា ឬដំណាក់ខ្លាំង}}។\\
ទីតាំងសូរអតិបរមា៖
$d_2-d_1=n\lambda=\Delta d = n\lambda$
ដែល $\lambda=\frac{v}{f}$\\
យើងបាន៖
$\boxed{d_2-d_1=\Delta d=\frac{nv}{f}}$

\khmersubtopic{២.២. ទីតាំងសូរអប្បបរមា (ដំណាក់ស្ងប់)}
