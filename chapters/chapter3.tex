\khmertopic{១. មេដែកនឹងដែនម៉ាញេទិក}
\khmersubtopic{១.១. មេដែក}
\noindent\hspace*{0.6cm}\faCaretRight~\textbf{\textcolor{brownheader}{មេដែក}} គឺជាអង្គធាតុដែលអាចឆក់ទាញដែកបាន។
លក្ខណៈឆក់ទាញទាំងនោះហៅថា \textbf{លក្ខណៈម៉ាញេទិក}
។ ជាទូទៅ មេដែកមានប៉ូលចំនួនពីរគឺ
ចុងម្ខាងជាប៉ូលជើង ({\englishfont N})
និងចុងម្ខាងជាប៉ូលជើង ({\englishfont S})។
ប៉ូលទាំងមិនអាចកាត់ផ្តាច់ពីគ្នាបានឡើយ
ហើយគេចែកវាជា២ប្រភេទគឺ៖

\noindent
\begin{minipage}[t]{0.55\textwidth}
	\raggedright
	\begin{itemize}[label={\textcolor{brownheader}{\faCaretRight}}]
		\item \textbf{មេដែកធម្មជាតិ} ជាមេដែកដែលមានស្រាប់នៅក្នុងធម្មជាតិដូចជា សំណាក \ce{Fe3O4} និងមេដែកផែនដី សញ្ញាជាដើម។
	\begin{itemize}[label={\textcolor{brownheader}{\faCaretRight}}]
		\item \textbf{មេដែកសិប្បនិម្មិត} ជាមេដែកដែលត្រូវបានបង្កើតឡើងតាមតម្រូវការចាំបាច់ មានដូចជា របាមេដែក ម្ជុលមេដែក និងមេដែករាងអក្សរ ({\englishfont U})។
	\end{itemize}
	\end{itemize}
	\khmersubsubtopic{ក. អន្តរកម្មនៃមេដែក}
	នៅពេលដែលគេដាក់មេដែកពីរជិតគ្នា
	នោះបាតុភូតពីរនឹងកើតឡើង
	នោះគឺមេដែកច្រានចេញពីគ្នា ឬទាញគ្នាចូល
	ដែលបាតុភូតនេះហៅថា
	\textbf{អន្តរកម្មនៃមេដែក ឬអន្តរកម្មមេដែក}
	និងមានលក្ខណៈដូចខាងក្រោម៖
	\begin{itemize}[label={\textcolor{brownheader}{\faCaretRight}}]
		\item បើគេដាក់ប៉ូលដូចគ្នា ({\englishfont N-N} ឬ {\englishfont S-S}) ជិតគ្នា នោះមេដែកទាំងពីរនឹងច្រានចេញពីគ្នា។
		\item បើគេដាក់ប៉ូលខុសគ្នា ({\englishfont N-S} ឬ {\englishfont S-N}) ជិតគ្នា នោះមេដែកទាំងពីរនឹងទាញគ្នាចូល។
	\end{itemize}

\khmersubtopic{១. ២. ដែនម៉ាញេទិកនៃមេដែក}
លក្ខណៈទាញគ្នាចូលនិងច្រានចេញនៃមេដែកគឺបានបង្កើតឡើងដោយសារកម្លាំងអន្តរកម្មម៉ាញេទិកនៃមេដែកទាំងនោះ។ កាលពីថ្នាក់ទី១១ យើងបានសិក្សារួចមកហើយថា កម្លាំងអគ្គិសនីដែលមានអំពើលើបន្ទុកអគ្គិសនីត្រូវបានពណ៌នាដោយលំហនៃដែនអគ្គិសនីជុំវិញបន្ទុកអគ្គិសនីស្ថិតនៅនឹង ហើយបង្កើតឲ្យមានអន្តរកម្មរវាងដែនអគ្គិសនីនិងបន្ទុកអគ្គិសនីនោះ។ នៅក្នុងការសិក្សាអំពីកម្លាំងម៉ាញេទិកក៏មានលក្ខណៈស្រដៀងទៅនឹងកម្លាំងអគ្គិសនីដែរ គឺមានន័យថាកម្លាំងម៉ាញេទិកដែលត្រូវបានបង្កើតឡើងដោយលំហនៃដែនម៉ាញេទិក មានអន្តរកម្មលើមេដែកដែលស្ថិតនៅក្នុងលំហនោះ។ ក្នុងការសិក្សាដែនម៉ាញេទិក គ៏មានលក្ខណៈស្រដៀងទៅនឹងកម្លាំងអគ្គិសនីដែរ គឺមានន័យថាកម្លាំងម៉ាញេទិក ដែលបង្កើតឡើងនៅក្នុងលំហនៃដែនម៉ាញេទិកមានអន្តរកម្មលើមេដែកស្ថិតនៅក្នងលំហនោះ។ ដូចនេះ ដែនម៉ាញេទិក ($\vec{B}$) គឺជាមជ្ឍដ្ឋានរូបធាតុពិសេសសដែលកើតឡើងជុំវិញមេដែក ហើយនឹងបង្កើតកម្លាំងម៉ាញេទិកលើមេដែក (ច្រានចេញ និងទាញចូល) ស្ថិតនៅក្នុងមជ្ឈដ្ឋានមួយនេះ។\\

\textbf{ករណីសំខាន់ៗនៃគន្លងរបស់ភាគល្អិតផ្លាស់ទីចូលក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន៖}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item ករណី $\theta = 0^{\circ}$ ($\vec{v}\nearrow\nearrow \vec{B}$) ឬ $\theta = 180^{\circ}$ ($\vec{v}\nearrow\swarrow \vec{B}$)៖ ភាគល្អិតផ្លាស់ទីនៅក្នុងដែនម៉ាញេទិកដោយរក្សាគន្លងដើម ដោយមិនរងកម្លាំងម៉ាញេទិក $F_{B(\min)} = 0$ N។
	\item ករណី $\theta = 90^{\circ}$ ($\vec{v}\perp\vec{B}$)៖ ភាគល្អិតផ្លាស់ទីនៅក្នុងដែនម៉ាញេទិកដោយចលនាវង់ស្មើ មានកាំ $R$ (ដូចរូបទី១៣.ក) ដោយរងនូវកម្លាំងម៉ាញេទិកពេញទិសអតិបរមា $F_{B(\max)} = |q|vB$។
	\item ករណី $\theta \neq 0^{\circ},\, 90^{\circ},\, 180^{\circ}$៖ ភាគល្អិតផ្លាស់ទីបង្កើតបានជាស្ប៉ៃ (ដូចរូបទី១៣.ខ) និងរងនូវកម្លាំងម៉ាញេទិក $F_B = |q|vB\sin\theta$ ហើយមានល្បឿន $\boxed{v = \sqrt{v_\parallel^2 + v_\perp^2}}$ (៦)។
\end{itemize}

\vspace{0.3em}
ដោយ $\sum F = F_{B(\max)} = |q|vB$ យើងបាន $|q|vB = m\dfrac{v^2}{R}$។

\vspace{0.4em}
\begin{center}
	$F_B = q\,v_d\,B\sin\theta$
\end{center}

\end{center}

\vspace{0.4em}
\begin{center}

	\item \textbf{ទិសដៅ} ៖ វិធានដៃស្តាំ (ដាក់ម្រាមទាំងបួនលាតត្រង់តាមទិសដៅចរន្ត $I$ ក្តាប់តាម $\vec{B}$ រីឯមេដៃកែងបង្ហាញពីទិសដៅនៃកម្លាំងម៉ាញេទិក $\vec{F}_B$)

	នៅពេលដែលភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីបាញ់ចូលក្នុងដែនម៉ាញេទិកត្រង់ចំណុច $O$ ដោយល្បឿនដើម $v_0$ ដែលមានទិសដៅកែងនឹងដែនម៉ាញេទិក $\vec{B}$ បន្ទាប់មកវាធ្វើចលនាក្នុងគន្លងកោង និងរងកម្លាំងម៉ាញេទិក $F_B = |q|vB$។

	\vspace{0.5em}
	តាមច្បាប់ទី២របស់ញូតុន ៖ $\sum\vec{F} = m\vec{a}$។

	\vspace{0.3em}
	ដោយ $\sum F = F_B = |q|vB$ និង $\vec{a} = \vec{a}_T + \vec{a}_N$ ។

	\vspace{0.3em}
	តែ $a_T = \dfrac{dv}{dt} = 0\ \mathrm{m/s^2}$ (ព្រោះ $v$ ថេរ) និង $a_N = \dfrac{v^2}{R}$ នាំឲ្យ៖ $a = a_N = \dfrac{v^2}{R}$។

	\vspace{0.3em}
	យើងបាន ៖ $|q|vB = m\dfrac{v^2}{R}$ ដូចនេះ $R = \dfrac{mv}{|q|B}$។
\end{minipage}%
\hfill
\begin{minipage}[t]{0.40\textwidth}
	\vspace{0pt}
	\hspace*{0.5em}%
	\khmerphoto[0.95\linewidth]{
		img/deflexion.jpg}{
		រូបទី១៨៖ លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន}
\end{minipage}

\vspace{0.5em}
\noindent
ដូចនេះ ភាគល្អិតធ្វើចលនាក្នុងគន្លងរង្វង់ដោយល្បឿនថេរ ដែលមានកាំ $R = \dfrac{mv}{|q|B}$ (ដូចកន្សោម (៧))។

\vspace{0.9em}
\khmersubtopic{៤. ២. លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីក្នុងដែនម៉ាញេទិក}

\noindent
តាមរូប ៖ $\tan\alpha = \dfrac{l}{R}$។

\vspace{0.3em}
ដោយមុំ $\alpha$ តូច យើងមាន $\tan\alpha \approx \alpha$ ($\mathrm{rad}$)។

\vspace{0.3em}
នាំឲ្យ ៖ $\alpha\ (\mathrm{rad}) = \dfrac{l}{R} = \dfrac{l}{mv/|q|B} = \dfrac{|q|Bl}{mv}$។

\vspace{0.7em}
\noindent
ដូចនេះ លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន គឺ៖

\vspace{0.4em}
\begin{center}
	$\boxed{\alpha\ (\mathrm{rad}) = \dfrac{|q|Bl}{mv}}$ \quad (១០)
\end{center}

\vspace{0.9em}
\noindent
\defbox{
\begin{center}
	$\alpha$ ៖ លំអៀងរបស់ភាគល្អិត ($\mathrm{rad}$)\\[1pt]
	$q$ ៖ បន្ទុកអគ្គិសនីរបស់ភាគល្អិត ($\mathrm{C}$)\\[1pt]
	$B$ ៖ អាំងតង់ស៊ីតេដែនម៉ាញេទិក ($\mathrm{T}$)\\[1pt]
	$l$ ៖ ប្រវែងដែនម៉ាញេទិកដែលភាគល្អិតឆ្លងកាត់ ($\mathrm{m}$)\\[1pt]
	$m$ ៖ ម៉ាសរបស់ភាគល្អិត ($\mathrm{kg}$)\\[1pt]
	$v$ ៖ ល្បឿនរបស់ភាគល្អិត ($\mathrm{m/s}$)
\end{center}
}

\vspace{0.9em}
\khmersubtopic{៤. ៣. ឌីផ្លិចស្យុងម៉ាញេទិក}

\noindent
តាមរូប ៖ $\tan\alpha \approx \alpha = \dfrac{Z}{D}$។

\vspace{0.3em}
នាំឲ្យ ៖ $Z = \alpha D = \dfrac{|q|Bl}{mv}\,D = \dfrac{|q|BlD}{mv}$។

\vspace{0.7em}
\noindent
ដូចនេះ ឌីផ្លិចស្យុងម៉ាញេទិករបស់ភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនី គឺ៖

\vspace{0.4em}
\begin{center}
	$\boxed{Z = \alpha D = \dfrac{|q|BlD}{mv}}$ \quad (១១)
\end{center}

\vspace{0.9em}
\noindent
\defbox{
\begin{center}
	$Z$ ៖ ឌីផ្លិចស្យុងម៉ាញេទិក ($\mathrm{m}$)\\[1pt]
	$D$ ៖ ចម្ងាយពីដែនម៉ាញេទិកដល់អេក្រង់ ($\mathrm{m}$)\\[1pt]
	$q$ ៖ បន្ទុកអគ្គិសនីរបស់ភាគល្អិត ($\mathrm{C}$)\\[1pt]
	$B$ ៖ អាំងតង់ស៊ីតេដែនម៉ាញេទិក ($\mathrm{T}$)\\[1pt]
	$l$ ៖ ប្រវែងដែនម៉ាញេទិកដែលភាគល្អិតឆ្លងកាត់ ($\mathrm{m}$)\\[1pt]
	$m$ ៖ ម៉ាសរបស់ភាគល្អិត ($\mathrm{kg}$)\\[1pt]
	$v$ ៖ ល្បឿនរបស់ភាគល្អិត ($\mathrm{m/s}$)
\end{center}
}

\khmersubtopic{៤. ៤. ស្ប៉ិចក្រាប ({\englishfont Mass Spectrograph})}

\noindent
\begin{minipage}[t]{0.55\textwidth}
	\raggedright
	គេប្រើវិធីសាស្ត្រមួយដើម្បីកំណត់ម៉ាស់របស់ភាគល្អិត ដោយប្រើឧបករណ៍មួយឈ្មោះថាស្ប៉ិចក្រាប (ដូចរូបទី១៩)។ ដំបូង ដោយឲ្យភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីដាច់ចេញពីបន្ទះលោហៈដោយគ្មានល្បឿនដើម ហើយបង្កើនល្បឿនបាន $v$ ក្រោមតង់ស្យុង $V$ ដែលបង្កើតដែនអគ្គិសនីចន្លោះបន្ទះលោហៈ។ បន្ទាប់មក ភាគល្អិតនោះបានចេញផុតពីដែនអគ្គិសនី និងផ្លាស់ទីក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន $\vec{B}$ ដោយគន្លងរង្វង់ស្មើមានកាំ $R = \dfrac{mv}{|q|B}$។

	\end{minipage}%
\hfill
\begin{minipage}[t]{0.40\textwidth}
	\vspace{0pt}
	\hspace*{0.5em}%
	\khmerphoto[0.95\linewidth]{
		img/mass_spectrometer.jpg}{
		រូបទី១៩៖ ស្ប៉ិចក្រាប ({\englishfont Mass Spectrograph})}
\end{minipage}

\vspace{0.6em}
\noindent\hspace*{2.5em}តាមច្បាប់រក្សាថាមពល ៖ $W = \Delta K$ ដោយ $W = qV$ និង $\Delta K = \tfrac{1}{2}mv^2$។

\noindent\hspace*{2.5em}ព្រោះ $\Delta K = K_f - K_i = \tfrac{1}{2}mv^2 - \tfrac{1}{2}mv_0^2 = \tfrac{1}{2}mv^2$ (ព្រោះ $v_0 = 0$)។

\vspace{0.4em}
\begin{flalign*}
	\hspace{2.5em}\text{យើងបាន ៖ } qV &= \tfrac{1}{2}mv^2\\
	\hspace{2.5em}\text{ប៉ុន្តែ ៖ } R &= \dfrac{mv}{|q|B} \qquad \text{នាំឲ្យ ៖ } v = \dfrac{R|q|B}{m}\\
	\hspace{2.5em}\text{នោះ ៖ } qV &= \tfrac{1}{2}m\left(\dfrac{R|q|B}{m}\right)^2\\
	\hspace{2.5em}m &= \dfrac{qR^2B^2}{2V} = \dfrac{q(RB)^2}{2V}
\end{flalign*}

\noindent
ដូចនេះ ម៉ាសដែលត្រូវបានវាស់ដោយស្ប៉ិចក្រាប គឺ៖

\vspace{0.4em}
\begin{center}
	$\boxed{m = \dfrac{q(RB)^2}{2V}}$ \quad (១២)
\end{center}

\vspace{0.9em}
\noindent
\defbox{
\begin{center}
	$m$ ៖ ម៉ាសរបស់ភាគល្អិត ($\mathrm{kg}$)\\[1pt]
	$q$ ៖ បន្ទុកអគ្គិសនីរបស់ភាគល្អិត ($\mathrm{C}$)\\[1pt]
	$R$ ៖ កាំគន្លងនៅក្នុងដែនម៉ាញេទិក ($\mathrm{m}$)\\[1pt]
	$B$ ៖ អាំងតង់ស៊ីតេដែនម៉ាញេទិក ($\mathrm{T}$)\\[1pt]
	$V$ ៖ តង់ស្យុងដែលបង្កើនល្បឿនភាគល្អិត ($\mathrm{V}$)
\end{center}
}
