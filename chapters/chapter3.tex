\khmertopic{១. មេដែកនឹងដែនម៉ាញេទិក}
\khmersubtopic{១.១. មេដែក}
\noindent\hspace*{0.6cm}\faCaretRight~\textbf{\textcolor{brownheader}{មេដែក}} គឺជាអង្គធាតុដែលអាចឆក់ទាញដែកបាន។
លក្ខណៈឆក់ទាញទាំងនោះហៅថា \textbf{លក្ខណៈម៉ាញេទិក}
។ ជាទូទៅ មេដែកមានប៉ូលចំនួនពីរគឺ
ចុងម្ខាងជាប៉ូលជើង ({\englishfont N})
និងចុងម្ខាងជាប៉ូលជើង ({\englishfont S})។
ប៉ូលទាំងមិនអាចកាត់ផ្តាច់ពីគ្នាបានឡើយ
ហើយគេចែកវាជា២ប្រភេទគឺ៖

\noindent
\begin{minipage}[t]{0.55\textwidth}
	\raggedright
	\begin{itemize}[label={\textcolor{brownheader}{\faCaretRight}}]
		\item \textbf{មេដែកធម្មជាតិ} ជាមេដែកដែលមានស្រាប់នៅក្នុងធម្មជាតិដូចជា សំណាក \ce{Fe3O4} និងមេដែកផែនដី សញ្ញាជាដើម។
	\begin{itemize}[label={\textcolor{brownheader}{\faCaretRight}}]
		\item \textbf{មេដែកសិប្បនិម្មិត} ជាមេដែកដែលត្រូវបានបង្កើតឡើងតាមតម្រូវការចាំបាច់ មានដូចជា របាមេដែក ម្ជុលមេដែក និងមេដែករាងអក្សរ ({\englishfont U})។
	\end{itemize}
	\end{itemize}
	\khmersubsubtopic{ក. អន្តរកម្មនៃមេដែក}
	នៅពេលដែលគេដាក់មេដែកពីរជិតគ្នា
	នោះបាតុភូតពីរនឹងកើតឡើង
	នោះគឺមេដែកច្រានចេញពីគ្នា ឬទាញគ្នាចូល
	ដែលបាតុភូតនេះហៅថា
	\textbf{អន្តរកម្មនៃមេដែក ឬអន្តរកម្មមេដែក}
	និងមានលក្ខណៈដូចខាងក្រោម៖
	\begin{itemize}[label={\textcolor{brownheader}{\faCaretRight}}]
		\item បើគេដាក់ប៉ូលដូចគ្នា ({\englishfont N-N} ឬ {\englishfont S-S}) ជិតគ្នា នោះមេដែកទាំងពីរនឹងច្រានចេញពីគ្នា។
		\item បើគេដាក់ប៉ូលខុសគ្នា ({\englishfont N-S} ឬ {\englishfont S-N}) ជិតគ្នា នោះមេដែកទាំងពីរនឹងទាញគ្នាចូល។
	\end{itemize}

\khmersubtopic{១. ២. ដែនម៉ាញេទិកនៃមេដែក}
លក្ខណៈទាញគ្នាចូលនិងច្រានចេញនៃមេដែកគឺបានបង្កើតឡើងដោយសារកម្លាំងអន្តរកម្មម៉ាញេទិកនៃមេដែកទាំងនោះ។ កាលពីថ្នាក់ទី១១ យើងបានសិក្សារួចមកហើយថា កម្លាំងអគ្គិសនីដែលមានអំពើលើបន្ទុកអគ្គិសនីត្រូវបានពណ៌នាដោយលំហនៃដែនអគ្គិសនីជុំវិញបន្ទុកអគ្គិសនីស្ថិតនៅនឹង ហើយបង្កើតឲ្យមានអន្តរកម្មរវាងដែនអគ្គិសនីនិងបន្ទុកអគ្គិសនីនោះ។ នៅក្នុងការសិក្សាអំពីកម្លាំងម៉ាញេទិកក៏មានលក្ខណៈស្រដៀងទៅនឹងកម្លាំងអគ្គិសនីដែរ គឺមានន័យថាកម្លាំងម៉ាញេទិកដែលត្រូវបានបង្កើតឡើងដោយលំហនៃដែនម៉ាញេទិក មានអន្តរកម្មលើមេដែកដែលស្ថិតនៅក្នុងលំហនោះ។ ក្នុងការសិក្សាដែនម៉ាញេទិក គ៏មានលក្ខណៈស្រដៀងទៅនឹងកម្លាំងអគ្គិសនីដែរ គឺមានន័យថាកម្លាំងម៉ាញេទិក ដែលបង្កើតឡើងនៅក្នុងលំហនៃដែនម៉ាញេទិកមានអន្តរកម្មលើមេដែកស្ថិតនៅក្នងលំហនោះ។ ដូចនេះ ដែនម៉ាញេទិក ($\vec{B}$) គឺជាមជ្ឍដ្ឋានរូបធាតុពិសេសសដែលកើតឡើងជុំវិញមេដែក ហើយនឹងបង្កើតកម្លាំងម៉ាញេទិកលើមេដែក (ច្រានចេញ និងទាញចូល) ស្ថិតនៅក្នុងមជ្ឈដ្ឋានមួយនេះ។\\

\textbf{ករណីសំខាន់ៗនៃគន្លងរបស់ភាគល្អិតផ្លាស់ទីចូលក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន៖}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item ករណី $\theta = 0^{\circ}$ ($\vec{v}\nearrow\nearrow \vec{B}$) ឬ $\theta = 180^{\circ}$ ($\vec{v}\nearrow\swarrow \vec{B}$)៖ ភាគល្អិតផ្លាស់ទីនៅក្នុងដែនម៉ាញេទិកដោយរក្សាគន្លងដើម ដោយមិនរងកម្លាំងម៉ាញេទិក $F_{B(\min)} = 0$ N។
	\item ករណី $\theta = 90^{\circ}$ ($\vec{v}\perp\vec{B}$)៖ ភាគល្អិតផ្លាស់ទីនៅក្នុងដែនម៉ាញេទិកដោយចលនាវង់ស្មើ មានកាំ $R$ (ដូចរូបទី១៣.ក) ដោយរងនូវកម្លាំងម៉ាញេទិកពេញទិសអតិបរមា $F_{B(\max)} = |q|vB$។
	\item ករណី $\theta \neq 0^{\circ},\, 90^{\circ},\, 180^{\circ}$៖ ភាគល្អិតផ្លាស់ទីបង្កើតបានជាស្ប៉ៃ (ដូចរូបទី១៣.ខ) និងរងនូវកម្លាំងម៉ាញេទិក $F_B = |q|vB\sin\theta$ ហើយមានល្បឿន $\boxed{v = \sqrt{v_\parallel^2 + v_\perp^2}}$ (៦)។
\end{itemize}

\vspace{0.3em}
ដោយ $\sum F = F_{B(\max)} = |q|vB$ យើងបាន $|q|vB = m\dfrac{v^2}{R}$។

	\item ល្បឿនតាមអ័ក្សស្របនឹង $\vec{B}$៖ $v_\parallel = v\cos\theta$
	\item ល្បឿនតាមអ័ក្សកែងនឹង $\vec{B}$៖ $v_\perp = v\sin\theta$
\end{itemize}

\begin{center}

\end{center}

\vspace{0.4em}
\begin{center}

\end{center}

ដូចនេះកន្សោមវ៉ិចទ័រគឺ៖

\vspace{0.4em}
\begin{center}
	$\vec{F}_B = I\,\vec{L}\times\vec{B}$
\end{center}

	\item \textbf{ទិសដៅ} ៖ វិធានដៃស្តាំ (ដាក់ម្រាមទាំងបួនលាតត្រង់តាមទិសដៅចរន្ត $I$ ក្តាប់តាម $\vec{B}$ រីឯមេដៃកែងបង្ហាញពីទិសដៅនៃកម្លាំងម៉ាញេទិក $\vec{F}_B$)

	\item ជ្រុង $SP$ និង $QR$ បានរងនូវកម្លាំង $\vec{F}_{SP}$ និង $\vec{F}_{QR}$ មានទិសដៅផ្ទុយគ្នា និងតម្លៃស្មើគ្នា គឺ $F_{SP} = F_{QR} = ILB$ នាំឲ្យកម្លាំងផ្គួបរវាងជ្រុងទាំងពីរមានតម្លៃមិនសូន្យ ព្រោះកម្លាំងទាំងពីរមានចំណុចចាប់មិននៅលើអ័ក្សតែមួយ ហើយបង្កើតឲ្យមានជាម៉ូម៉ង់នៃកម្លាំងរង្វិលនៃស៊ុមនេះ។
\end{itemize}

\khmertopic{៤. លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីនៅក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន}

\khmersubtopic{៤. ១. ប្រភេទចលនារបស់ភាគល្អិត}

\noindent
\begin{minipage}[t]{0.55\textwidth}
	\raggedright
	នៅពេលដែលភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីបាញ់ចូលក្នុងដែនម៉ាញេទិកត្រង់ចំណុច $O$ ដោយល្បឿនដើម $v_0$ ដែលមានទិសដៅកែងនឹងដែនម៉ាញេទិក $\vec{B}$ បន្ទាប់មកវាធ្វើចលនាក្នុងគន្លងកោង និងរងកម្លាំងម៉ាញេទិក $F_B = |q|vB$។

	\vspace{0.5em}
	តាមច្បាប់ទី២របស់ញូតុន ៖ $\sum\vec{F} = m\vec{a}$។

	\vspace{0.3em}
	ដោយ $\sum F = F_B = |q|vB$ និង $\vec{a} = \vec{a}_T + \vec{a}_N$ ។

	\vspace{0.3em}
	តែ $a_T = \dfrac{dv}{dt} = 0\ \mathrm{m/s^2}$ (ព្រោះ $v$ ថេរ) និង $a_N = \dfrac{v^2}{R}$ នាំឲ្យ៖ $a = a_N = \dfrac{v^2}{R}$។

	\vspace{0.3em}
	យើងបាន ៖ $|q|vB = m\dfrac{v^2}{R}$ ដូចនេះ $R = \dfrac{mv}{|q|B}$។
\end{minipage}%
\hfill
\begin{minipage}[t]{0.40\textwidth}
	\vspace{0pt}
	\hspace*{0.5em}%
	\khmerphoto[0.95\linewidth]{
		img/deflexion.jpg}{
		រូបទី១៨៖ លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន}
\end{minipage}

\vspace{0.5em}
\noindent
ដូចនេះ ភាគល្អិតធ្វើចលនាក្នុងគន្លងរង្វង់ដោយល្បឿនថេរ ដែលមានកាំ $R = \dfrac{mv}{|q|B}$ (ដូចកន្សោម (៧))។

\vspace{0.9em}
\khmersubtopic{៤. ២. លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីក្នុងដែនម៉ាញេទិក}

\noindent
តាមរូប ៖ $\tan\alpha = \dfrac{l}{R}$។

\vspace{0.3em}
ដោយមុំ $\alpha$ តូច យើងមាន $\tan\alpha \approx \alpha$ ($\mathrm{rad}$)។

\vspace{0.3em}
នាំឲ្យ ៖ $\alpha\ (\mathrm{rad}) = \dfrac{l}{R} = \dfrac{l}{mv/|q|B} = \dfrac{|q|Bl}{mv}$។

\vspace{0.7em}
\noindent
ដូចនេះ លំអៀងនៃភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន គឺ៖

\vspace{0.4em}
\begin{center}
	$\boxed{\alpha\ (\mathrm{rad}) = \dfrac{|q|Bl}{mv}}$ \quad (១០)
\end{center}

\vspace{0.9em}
\noindent
\defbox{
\begin{center}
	$\alpha$ ៖ លំអៀងរបស់ភាគល្អិត ($\mathrm{rad}$)\\[1pt]
	$q$ ៖ បន្ទុកអគ្គិសនីរបស់ភាគល្អិត ($\mathrm{C}$)\\[1pt]
	$B$ ៖ អាំងតង់ស៊ីតេដែនម៉ាញេទិក ($\mathrm{T}$)\\[1pt]
	$l$ ៖ ប្រវែងដែនម៉ាញេទិកដែលភាគល្អិតឆ្លងកាត់ ($\mathrm{m}$)\\[1pt]
	$m$ ៖ ម៉ាសរបស់ភាគល្អិត ($\mathrm{kg}$)\\[1pt]
	$v$ ៖ ល្បឿនរបស់ភាគល្អិត ($\mathrm{m/s}$)
\end{center}
}

\vspace{0.9em}
\khmersubtopic{៤. ៣. ឌីផ្លិចស្យុងម៉ាញេទិក}

\noindent
តាមរូប ៖ $\tan\alpha \approx \alpha = \dfrac{Z}{D}$។

\vspace{0.3em}
នាំឲ្យ ៖ $Z = \alpha D = \dfrac{|q|Bl}{mv}\,D = \dfrac{|q|BlD}{mv}$។

\vspace{0.7em}
\noindent
ដូចនេះ ឌីផ្លិចស្យុងម៉ាញេទិករបស់ភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនី គឺ៖

\vspace{0.4em}
\begin{center}
	$\boxed{Z = \alpha D = \dfrac{|q|BlD}{mv}}$ \quad (១១)
\end{center}

\vspace{0.9em}
\noindent
\defbox{
\begin{center}
	$Z$ ៖ ឌីផ្លិចស្យុងម៉ាញេទិក ($\mathrm{m}$)\\[1pt]
	$D$ ៖ ចម្ងាយពីដែនម៉ាញេទិកដល់អេក្រង់ ($\mathrm{m}$)\\[1pt]
	$q$ ៖ បន្ទុកអគ្គិសនីរបស់ភាគល្អិត ($\mathrm{C}$)\\[1pt]
	$B$ ៖ អាំងតង់ស៊ីតេដែនម៉ាញេទិក ($\mathrm{T}$)\\[1pt]
	$l$ ៖ ប្រវែងដែនម៉ាញេទិកដែលភាគល្អិតឆ្លងកាត់ ($\mathrm{m}$)\\[1pt]
	$m$ ៖ ម៉ាសរបស់ភាគល្អិត ($\mathrm{kg}$)\\[1pt]
	$v$ ៖ ល្បឿនរបស់ភាគល្អិត ($\mathrm{m/s}$)
\end{center}

\khmersubtopic{៤. ៤. ស្ប៉ិចក្រាប ({\englishfont Mass Spectrograph})}

\noindent
\begin{minipage}[t]{0.55\textwidth}
	\raggedright
	គេប្រើវិធីសាស្ត្រមួយដើម្បីកំណត់ម៉ាស់របស់ភាគល្អិត ដោយប្រើឧបករណ៍មួយឈ្មោះថាស្ប៉ិចក្រាប (ដូចរូបទី១៩)។ ដំបូង ដោយឲ្យភាគល្អិតផ្ទុកបន្ទុកអគ្គិសនីដាច់ចេញពីបន្ទះលោហៈដោយគ្មានល្បឿនដើម ហើយបង្កើនល្បឿនបាន $v$ ក្រោមតង់ស្យុង $V$ ដែលបង្កើតដែនអគ្គិសនីចន្លោះបន្ទះលោហៈ។ បន្ទាប់មក ភាគល្អិតនោះបានចេញផុតពីដែនអគ្គិសនី និងផ្លាស់ទីក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន $\vec{B}$ ដោយគន្លងរង្វង់ស្មើមានកាំ $R = \dfrac{mv}{|q|B}$។

	\end{minipage}%
\hfill
\begin{minipage}[t]{0.40\textwidth}
	\vspace{0pt}
	\hspace*{0.5em}%
	\khmerphoto[0.95\linewidth]{
		img/mass_spectrometer.jpg}{
		រូបទី១៩៖ ស្ប៉ិចក្រាប ({\englishfont Mass Spectrograph})}
\end{minipage}

\vspace{0.6em}
\noindent\hspace*{2.5em}តាមច្បាប់រក្សាថាមពល ៖ $W = \Delta K$ ដោយ $W = qV$ និង $\Delta K = \tfrac{1}{2}mv^2$។

\noindent\hspace*{2.5em}ព្រោះ $\Delta K = K_f - K_i = \tfrac{1}{2}mv^2 - \tfrac{1}{2}mv_0^2 = \tfrac{1}{2}mv^2$ (ព្រោះ $v_0 = 0$)។

\vspace{0.4em}
\begin{flalign*}
	\hspace{2.5em}\text{យើងបាន ៖ } qV &= \tfrac{1}{2}mv^2\\
	\hspace{2.5em}\text{ប៉ុន្តែ ៖ } R &= \dfrac{mv}{|q|B} \qquad \text{នាំឲ្យ ៖ } v = \dfrac{R|q|B}{m}\\
	\hspace{2.5em}\text{នោះ ៖ } qV &= \tfrac{1}{2}m\left(\dfrac{R|q|B}{m}\right)^2\\
	\hspace{2.5em}m &= \dfrac{qR^2B^2}{2V} = \dfrac{q(RB)^2}{2V}
\end{flalign*}

\noindent
ដូចនេះ ម៉ាសដែលត្រូវបានវាស់ដោយស្ប៉ិចក្រាប គឺ៖

\vspace{0.4em}
\begin{center}
	$\boxed{m = \dfrac{q(RB)^2}{2V}}$ \quad (១២)
\end{center}

\vspace{0.9em}
\noindent
\defbox{
\begin{center}
	$m$ ៖ ម៉ាសរបស់ភាគល្អិត ($\mathrm{kg}$)\\[1pt]
	$q$ ៖ បន្ទុកអគ្គិសនីរបស់ភាគល្អិត ($\mathrm{C}$)\\[1pt]
	$R$ ៖ កាំគន្លងនៅក្នុងដែនម៉ាញេទិក ($\mathrm{m}$)\\[1pt]
	$B$ ៖ អាំងតង់ស៊ីតេដែនម៉ាញេទិក ($\mathrm{T}$)\\[1pt]
	$V$ ៖ តង់ស្យុងដែលបង្កើនល្បឿនភាគល្អិត ($\mathrm{V}$)
\end{center}
}

	\raggedright
	លោកផារ៉ាដេបានដំឡើងសៀគ្វីពិសោធន៍មួយ (ដូចរូបទី២១) ដើម្បីរកចរន្តអាំងឌុចស្យុង។ ពេលបិទកុងតាក់ ធ្វើឱ្យចរន្តឆ្លងកាត់សៀគ្វីនៃបុំទី១ (ខាងឆ្វេង) និងធ្វើឱ្យកើតមានចរន្តឆ្លងកាត់សៀគ្វីនៃបុំទី២ តែក្នុងពេលចាប់ផ្ដើមបើក និងពេលបិទកុងតាក់ភ្ជាប់ (ចរន្តនេះកើតមានតែក្នុងពេលខ្លីបំផុត និងមានតម្លៃធំ)។ វាត្រូវបានបង្កើតឡើងដោយសារមានបម្រែបម្រួលដែនម៉ាញេទិកនៃសៀគ្វីនៃបុំទី១។

	បន្ទាត់ដែនម៉ាញេទិកដែលឆ្លងកាត់សៀគ្វីនៅក្នុងបុំទី២ ធ្វើឱ្យមានបម្រែបម្រួលភ្លុចម៉ាញេទិកឆ្លងកាត់បុំទី២។ ចរន្តដែលឆ្លងកាត់សៀគ្វីនៃបុំទី២ ជា \textbf{ចរន្តអាំងឌុចស្យុង} ({\englishfont Induced Current}) ដែលត្រូវបានបង្ហាញដោយអំពែម៉ែត្រ (Ammeter/Ampere meter) ហើយកម្មន្តដែលបានបំពេញក្នុងមួយខ្នាតបន្ទុក ដើម្បីធ្វើឱ្យអេឡិចត្រុងសេរីផ្លាស់ទី បង្កើតបានជាចរន្ត ត្រូវបានឲ្យឈ្មោះថា កម្លាំងអគ្គិសនីចលករអាំងឌុចស្យុង ({\englishfont Induced EMF})។ ចំពោះដំណើរការនៃការបង្កើតចរន្ត និងកម្លាំងអគ្គិសនីចលករនេះ ត្រូវបានឲ្យឈ្មោះថា អាំងឌុចស្យុង ({\englishfont Induction})។
\end{minipage}%

\khmersubtopic{១. ២. ពិសោធន៍ទី២របស់លោកផារ៉ាដេ}

	\begin{itemize}
		\item នៅពេលគេផ្លាស់ទីមេដែកយ៉ាងលឿនចូលទៅក្នុងរបុំខ្សែចម្លង ពេលនោះចរន្តអាំងឌុចស្យុងត្រូវបានបង្កើតឡើងនៅក្នុងរបុំខ្សែចម្លង ដែលបង្ហាញដោយកាល់វ៉ាណូម៉ែត្រ។
		\item នៅពេលដែលគេដាក់មេដែកនៅនឹងធៀបនឹងរបុំខ្សែចម្លង ពេលនោះរបុំមិនត្រូវបានបង្កើតចរន្តអាំងឌុចស្យុងទេ។
		\item នៅពេលគេទាញមេដែកផ្លាស់ទីចេញមកវិញ កាល់វ៉ាណូម៉ែត្រក៏ងាកផ្ទុយពីទិសដៅ នៅពេលដែលរុញចូល។
	\end{itemize}

\khmersubtopic{១. ៣. ភ្លុចម៉ាញេទិក និងភ្លុចអាំងឌុចស្យុង}

\vspace{0.7em}
\noindent
ភ្លុចម៉ាញេទិក ជាទំហំមួយសម្រាប់សម្គាល់ចំនួនខ្សែដែនម៉ាញេទិកឆ្លងកាត់ផ្ទៃណាមួយ (ដូចរូបទី៩.ក)។ ភ្លុចម៉ាញេទិក $\Phi_B$ ត្រូវបានកំណត់ដោយផលគុណស្កាឡែរវាងវ៉ិចទ័រដែនម៉ាញេទិក $(\vec{B})$ ជាមួយនឹងវ៉ិចទ័រផ្ទៃណាមួយ $(\vec{A})$៖

\begin{center}
	$\boxed{\Phi_B = \vec{B} \cdot \vec{A} = BA\cos(\vec{B}, \vec{A}) = B_\perp A}$ \quad (១)
\end{center}

\noindent
គេបានភ្លុចម៉ាញេទិកកំណត់ដោយ៖

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item $\Phi_B$ ៖ ភ្លុចម៉ាញេទិក ($\mathrm{Wb}$)
	\item $B_\perp = B\cos(\vec{B}, \vec{A})$ ៖ ដែនម៉ាញេទិកឆ្លងកាត់ផ្ទៃ និងមានទិសដៅកែងនឹងផ្ទៃ ($\mathrm{T}$)
	\item $A$ ៖ ផ្ទៃដែលត្រូវបានដែនម៉ាញេទិកឆ្លងកាត់ ($\mathrm{m}^2$)
	\item $(\vec{B}, \vec{A}) = \alpha$ ៖ មុំដែលបង្កើតដោយ $(\vec{B})$ និង $(\vec{A})$ ៖ ($\mathrm{rad}$ ឬ ដឺក្រេ)
\end{itemize}

\vspace{0.9em}
\noindent
\textbf{ករណីសំខាន់សម្រាប់ភ្លុចម៉ាញេទិកឆ្លងកាត់ផ្ទៃណាមួយ៖}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item ករណី $\alpha = 90^{\circ}$ ($\vec{B} \perp \vec{A}$)៖ ភ្លុចម៉ាញេទិកមានតម្លៃអតិបរមា $\Phi_{B(\max)} = BA$។
	\item ករណី $\alpha = 0^{\circ},\, 180^{\circ}$ ($\vec{B} \parallel \vec{A}$)៖ ភ្លុចម៉ាញេទិកសូន្យាស់ដោយរក្សាគន្លងដើម ដោយមិនរងភ្លុចម៉ាញេទិកទេ $\Phi_{B(\min)} = 0$។
\end{itemize}

\vspace{0.5em}
\khmerphoto[0.95\linewidth]{img/magflux.jpg}{រូបទី២៣៖ ភ្លុចម៉ាញេទិកឆ្លងកាត់ផ្ទៃ}

\begin{center}
	(ក) $\Phi_B = BA\cos(\alpha)$ \qquad\qquad
	(ខ) $\Phi_B = BA$ \qquad\qquad
	(គ) $\Phi_B = 0$
\end{center}

\begin{center}
	\textbf{Editor's Note:   }ផ្ទែ $\vec{A}$
	មិនមែនជាវុិចទ័រទេ។ អ្វីដែលអាចបង្កើតមុំ $\alpha$ បានគឺវុិចទ័រកៃងផ្ទៃ $\vec{n}$ នោះទេ។
\end{center}

\khmersubtopic{១. ៤. ច្បាប់ផារ៉ាដេ ({\englishfont Faraday's Law})}

\vspace{0.7em}
\noindent
\textit{កម្លាំងអគ្គិសនីចលករអាំងឌី្វ} គឺជាទំហំមួយដែលសមាមាត្រនឹងបម្រែបម្រួលភ្លុចឆ្លងកាត់សៀវ ឬរបុំខ្សែចម្លងណាមួយក្នុងមួយខ្នាតពេល។ កម្លាំងអគ្គិសនីចលករអាំងឌី្វ ត្រូវកំណត់តាងដោយ $E$ និងមានរូបមន្ត៖

\begin{center}
	$\boxed{E = -N\dfrac{\Delta\Phi_B}{\Delta t}}$ \quad (២)
\end{center}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item $E$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌី្វ ($\mathrm{V}$)
	\item $N$ ៖ ចំនួនស្ពៀររបស់របុំ (ជុំឬស្ពៀ)
	\item $\Delta\Phi_B = \Phi_{B2} - \Phi_{B1} = B_2A_2\cos\alpha_2 - B_1A_1\cos\alpha_1$ ៖ បម្រែបម្រួលភ្លុចម៉ាញេទិក ($\mathrm{Wb}$)
	\item $\Delta t = t_2 - t_1$ ៖ បម្រែបម្រួលរយៈពេលនៃការប្រែបម្រួលភ្លុចម៉ាញេទិក ($\mathrm{s}$)
\end{itemize}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item[$\succ$] \textbf{កំណត់សម្គាល់៖} ប្រសិនបើការប្រែបម្រួលនៃភ្លុចនោះក្នុងរយៈពេលខ្លី $\Delta t \to 0$ ពេលនោះបម្រែបម្រួលនៃភ្លុចម៉ាញេទិកក៏មានតម្លៃតូចដែរ $\Delta\Phi_B \to 0$ កម្លាំងអគ្គិសនីចលករដែលបង្កើតនៅក្នុងរបុំនោះ ក្លាយទៅជាកម្លាំងអគ្គិសនីចលករខណៈ៖
\end{itemize}

\begin{center}
	$\boxed{e(t) = -N\dfrac{d\Phi_B}{dt}}$ \quad (៣)
\end{center}

\noindent
ដែល៖ $\dfrac{d\Phi_B}{dt}$ គឺជាដេរីវេនៃអនុគមន៍ $\phi = BA\cos\alpha$ ធៀបនឹងពេល។

\begin{center}
	$\succ$ \quad \textbf{
		\underline{ករណីសំខាន់សម្រាប់ការប្រែប្រួលភ្លុចម៉ាញេទិក៖}}
\end{center}

\khmersubtopic{១. ៥. កម្លាំងអគ្គិសនីចលករអាំងឌ្វីក្នុងខ្សែចម្លង}

\vspace{0.7em}
\noindent
\begin{minipage}[t]{0.55\textwidth}
	\raggedright
នៅពេលដែលមានរបារលោហៈផ្លាស់ទីក្នុងដែនម៉ាញេទិកនៅលើទម្រលោហៈរាង U ស្ថិតនៅក្នុងដែនម៉ាញេទិកដែលមានទិសដៅចូលក្នុងផ្ទៃ (ដូចរូបទី២៤) ធ្វើឱ្យផ្ទៃដែលឆ្លងកាត់ដោយដែនម៉ាញេទិកនោះប្រែប្រួលផ្ទៃ $(\Delta A = L\Delta x)$ ដែលនាំកើតមានបម្រែបម្រួលភ្លុចម៉ាញេទិក $(\Delta \Phi_B = B\Delta A)$ នៅក្នុងវង់បិទដែលបង្កើតឡើងដោយរបារលោហៈ និងទម្រលោហៈរាង U ។

	\vspace{0.2em}
	\begin{flalign*}
		\text{តាមច្បាប់ផារ៉ាដេ ៖ } |E| &= \dfrac{|\Delta\Phi_B|}{\Delta t}\\
		&= \dfrac{B\Delta A}{\Delta t}\\
		&= \dfrac{B \times L\Delta x}{\Delta t}
	\end{flalign*}

	\begin{center}
		$\boxed{|E| = Blv}$ \quad (៤)
	\end{center}

	\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
		\item $B$ ៖ ដែនម៉ាញេទិក $(\mathrm{T})$
		\item $L$ ៖ ប្រវែងរបារលោហៈ $(\mathrm{m})$
		\item $v$ ៖ ល្បឿនរបស់របារលោហៈផ្លាស់ទី $(\mathrm{m/s})$
		\item $E$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វី $(\mathrm{V})$
	\end{itemize}
\end{minipage}%

	\vspace{0pt}
	\khmerphoto[0.95\linewidth]{img/magneticrod.jpg}{រូបទី២៤៖ របារលោហៈផ្លាស់ទីនៅលើទម្រលោហៈ}
\end{minipage}

\vspace{0.7em}
\noindent
\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item[$\succ$] ដើម្បីគណនាចរន្តអាំងឌ្វីដែលកាត់នៅក្នុងសៀគ្វី យើងត្រូវប្រើច្បាប់អូម $V = |E| = RI$
\end{itemize}

		\item $R$ ៖ រេស៊ីស្តង់នៃសៀគ្វី $(\Omega)$

\khmertopic{២. ច្បាប់ឡិន (\englishfont Lenz's Law)}
\khmersubtopic{២. ១. ច្បាប់ឡិនទី១}
\textbf{\textcolor{brownheader}{ពំនោល៖}} "ចរន្តអាំងឌ្វីមានទិសដៅយ៉ាងណា ដើម្បីប្រឆាំងនឹងបុព្វហេតុអ្នកឱ្យកំណើតវា" ។
\textit{ឧទាហរណ៍៖} គេរុញមេដែក(ចាត់ទុកជាអ្នកផ្តល់កំណើតនៃចរន្តអាំងឌ្វី) ចូលទៅក្នុងរបុំខ្សែចម្លងមួយត្រូវបានភ្ជាប់ទៅនឹងអំពែម៉ែត្រ(ដូចបង្ហាញក្នុងរូបទី២៥) ។ នោះយើងត្រូវអនុវត្តវិធានបាតដៃស្តាំចំពោះរបុំ (ស្រមៃថារបុំនោះជាអ្នកបង្កើតដែនម៉ាញេទិកដែលមានប៉ូលដូចប៉ូលមេដែក ដើម្បីទាញរកទិសដៅចរន្ត) ។ ចរន្តអាំងឌ្វីត្រូវបានបង្កើតឡើងដោយរបុំផ្ទាល់ ដើម្បីប្រឆាំងមានទិសដៅផ្ទុយពីចរន្តដែលយើងទើបនឹងស្រមៃនោះ។

\textbf{\textcolor{brownheader}{ពំនោល៖}} “ចរន្តអាំងឌ្វីបង្កើតដែនម៉ាញេទិចថ្មីមួយ ដើម្បីប្រឆាំងនឹងបម្រែបម្រួលភ្លុចម៉ាញេទិចឆ្លងកាត់របុំខ្សែចម្លងនោះ” ។
\textit{ឧទាហរណ៍៖} គេមានមេដែក និងរបុំខ្សែចម្លង ដូចបានបង្ហាញក្នុងរូបទី២៦ នៅពេលដែលគេផ្លាស់ទីមេដែកចេញ ឬចូលក្នុងរបុំខ្សែចម្លង នាំឱ្យចរន្តអាំងឌ្វីបានបង្កើតដែនម៉ាញេទិចថ្មីមួយ ដើម្បីប្រឆាំងនឹងការកើនឡើង ឬថយចុះឆ្លងកាត់របុំខ្សែចម្លងនោះ (តាមករណីទាំង៤ដូចក្នុងរូបខាងក្រោម)។

\khmersubtopic{៤. ១. ជនិតាអគ្គិសនី (\textbf{\englishfont Electric Generators})}

\textbf{\textcolor{brownheader}{ជនិតាអគ្គិសនី}} គឺជាឧបករណ៍មួយដែលបំប្លែងថាមពលមេកានិចទៅថាមពលអគ្គិសនី។ តាមច្បាប់ផាវ៉ាដេ នៅពេលដែលគេផ្តល់ថាមពលមេកានិចដោយបង្វិលស៊ុម (ដូចបង្ហាញនៅក្នុងរូបទី២៨) បានធ្វើឱ្យមានបម្រែបម្រួលភ្លុចម៉ាញេទិកឆ្លងកាត់ស៊ុម (របុំខ្សែចម្លង)។ ដូចនេះកម្លាំងអគ្គិសនីចលករគឺ

យើងបានសិក្សារួចមកហើយពីច្បាប់អាំងឌុចស្យុងរបស់លោកផារ៉ាដេ បង្ហាញពីការបង្កើតកម្លាំងអគ្គិសនីចលករអាំងឌ្វីនៅក្នុងរបុំខ្សែចម្លងមួយ ដោយមានបម្រែបម្រួលភ្លុចម៉ាញេទិចឆ្លងកាត់របុំខ្សែចម្លងវា។ នៅក្នុងមេរៀននេះយើងនឹងសិក្សាធាតុក្ខុកអាំងឌុចស្យុងដែលនាំឱ្យបង្កើតកម្លាំងអគ្គិសនីចលករ នោះដោយពិនិត្យលើរបុំខ្សែចម្លងពីរដាក់នៅក្បែរគ្នា ដូចជានៅពេលដែលមានបម្រែបម្រួលចរន្តនៅក្នុងរបុំទី១ នាំឱ្យបង្កើតបានកម្លាំងអគ្គិសនីចលករនៅក្នុងរបុំទី២ ។ ទំនាក់ទំនងចន្លោះរវាងរបុំទាំងពីរនេះ ត្រូវបកស្រាយដោយ \textbf{\textit{អាំងឌុចតង់ផ្ទេរ}} ។ របុំខ្សែចម្លងទាំងពីរ នេះមានឈ្មោះថា \textbf{\textit{អាំងឌុចទ័រ}} ។ មួយវិញទៀត គឺសិក្សាទៅលើការបង្កើតបានកម្លាំងអគ្គិសនីចលករនៅក្នុងរបុំខ្សែចម្លង១ នៅពេលដែលមានបម្រែបម្រួលចរន្តឆ្លងកាត់របុំខ្សែចម្លង ឬអាំងឌុចទ័រខ្លួនឯងផ្ទាល់ ត្រូវបានបកស្រាយដោយ \textbf{\textit{អាំងឌុចតង់}} ឬ\textbf{\textit{អាំងឌុចតង់ផ្ទាល់}} ។ ជាចុងក្រោយគឺផ្តោតសំខាន់ទៅលើ \textbf{\textit{ថាមពលម៉ាញេទិច}} ដែលជាថាមពលដែលអាំងឌុចទ័រត្រូវបានឆ្លងកាត់ដោយចរន្តអគ្គិសនី ព្រមទាំងលក្ខណៈនៃអាំងឌុចទ័រនៅក្នុងសៀគ្វី ដូចជាសៀគ្វី $RL$ និងសៀគ្វី $LC$ ។

ចរន្តឆ្លងកាត់កាល់វ៉ាណូម៉ែត្រ (ដូចរូបទី៣០) ដែលអាស្រ័យនឹងបម្រែបម្រួលចរន្តនៅក្នុងរបុំខាងឆ្វេង។ ដោយផ្អែកលើទំនាក់ទំនងរវាងរបុំទាំង២ យើងទាញបានភ្លុចម៉ាញេទិកសរុបដែលឆ្លងកាត់របុំទី២ គឺសមាមាត្រនឹងចរន្តឆ្លងកាត់របុំទី១ $(N_2\Phi_2 \propto i_1)$ ដែលមានមេគុណសមាមាត្រ $M_{21}$ ។

យើងបាន ៖
\begin{center}
	$\boxed{N_2\Phi_2 = M_{21}i_1}$ \quad (៨)
\end{center}

នាំឱ្យ : $N_2 \dfrac{d\Phi_2}{dt} = M_{21} \dfrac{di_1}{dt}$ \quad យកជំនួសក្នុង (៧)

ដូចនេះកម្លាំងអគ្គិសនីចលករអាំងឌ្វីខណៈដែលបង្កើតនៅក្នុងរបុំទី២ គឺ
\begin{center}
	$\boxed{e_2 = -M_{21} \dfrac{di_1}{dt}}$ \quad (៩)
\end{center}

យើងទាញបានកម្លាំងអគ្គិសនីចលករអាំងឌ្វីមធ្យមបង្កើតនៅក្នុងរបុំទី២
\begin{center}
	$\boxed{E_2 = -M_{21} \dfrac{\Delta I_{(1)}}{\Delta t}}$ \quad (១០)
\end{center}

ក្នុងការប្រើប្រាស់មកវិញនៅពេលដែលរបុំទី២ត្រូវបានភ្ជាប់ទៅនឹងប្រភពនៃចរន្តអគ្គិសនីប្រែប្រួល $(i_2)$ យើងបាន៖ ភ្លុចសរុបដែលឆ្លងកាត់របុំទី១គឺ
\begin{center}
	$N_1\Phi_1 = M_{12}i_2$ \quad (១១)
\end{center}

កម្លាំងអគ្គិសនីចលករអាំងឌ្វីខណៈគឺ
\begin{center}
	$\boxed{e_1 = -M_{12} \dfrac{di_2}{dt}}$ \quad (១២)
\end{center}

កម្លាំងអគ្គិសនីចលករអាំងឌ្វីមធ្យមបង្កើតនៅក្នុងរបុំទី១ គឺ
\begin{center}
	$\boxed{E_1 = -M_{12} \dfrac{\Delta I_{(2)}}{\Delta t}}$ \quad (១៣)
\end{center}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item តាមរយៈការពិសោធន៍ ទោះបីជារបុំទាំងពីរមិនមានលក្ខណៈដូចគ្នា (\textit{identical}) ឬភ្លុចឆ្លងកាត់មិនមានតម្លៃដូចគ្នាក៏ដោយ គឺអាំងឌុចតង់ផ្ទេររវាងរបុំទាំងពីរមានតម្លៃថេរ $M_{12} = M_{21} = M$ ។
\end{itemize}

តាមសមីការ(៨) និង(១១) ដូចនេះអាំងឌុចតង់ផ្ទេរគឺ
\begin{center}
	$\boxed{M = \dfrac{N_1\Phi_1}{i_2} = \dfrac{N_2\Phi_2}{i_1}}$ \quad (១៤)
\end{center}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item $M$ ៖ អាំងឌុចតង់ផ្ទេរ $(\text{H})$
	\item $M_{12}$ ៖ អាំងឌុចតង់ផ្ទេររវាងរបុំទី២ ទៅរបុំទី១ $(\text{H})$
	\item $M_{21}$ ៖ អាំងឌុចតង់ផ្ទេររវាងរបុំទី១ ទៅរបុំទី២ $(\text{H})$
	\item $\Phi_1$ ៖ ភ្លុចម៉ាញេទិកដែលឆ្លងកាត់របុំទី១ $(\text{Wb})$
	\item $\Phi_2$ ៖ ភ្លុចម៉ាញេទិកដែលឆ្លងកាត់របុំទី២ $(\text{Wb})$
	\item $N_1$ ៖ ចំនួនស្ពៀនៃរបុំទី១ $(\text{ជុំ})$
	\item $N_2$ ៖ ចំនួនស្ពៀនៃរបុំទី២ $(\text{ជុំ})$
	\item $i_1$ ៖ ចរន្តឆ្លងកាត់របុំទី១ $(\text{A})$
	\item $i_2$ ៖ ចរន្តឆ្លងកាត់របុំទី២ $(\text{A})$
	\item $\Delta I_{(1)} = I_{2(1)} - I_{1(1)}$ ៖ បម្រែបម្រួលចរន្តឆ្លងកាត់របុំទី១ $(\text{A})$
	\item $\Delta I_{(2)} = I_{2(2)} - I_{1(2)}$ ៖ បម្រែបម្រួលចរន្តឆ្លងកាត់របុំទី២ $(\text{A})$
	\item $\Delta t = t_2 - t_1$ ៖ បម្រែបម្រួលរយៈពេល $(\text{s})$
	\item $\frac{di_1}{dt}$ ៖ អត្រាចរន្តឆ្លងកាត់របុំទី១ $(\text{A/s})$
	\item $\frac{di_2}{dt}$ ៖ អត្រាចរន្តឆ្លងកាត់របុំទី២ $(\text{A/s})$
	\item $e_1$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីខណៈនៃរបុំទី១ $(\text{V})$
	\item $e_2$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីខណៈនៃរបុំទី២ $(\text{V})$
	\item $E_1$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីមធ្យមនៃរបុំទី១ $(\text{V})$
	\item $E_2$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីមធ្យមនៃរបុំទី២ $(\text{V})$
\end{itemize}

	\vspace{0pt}
	\khmerphoto[0.95\linewidth]{img/magneticfluxviasolenoid.png}{រូបទី៣៣៖ បូប៊ីនឆ្លងកាត់ដោយចរន្ត $i$ បង្កើតភ្លុចម៉ាញេទិចឆ្លងកាត់វាផ្ទាល់}

យើងបាន ៖
\begin{center}
	$\boxed{N\Phi = Li}$ \quad (១៥)
\end{center}

\begin{flalign*}
	\text{នាំឱ្យ ៖ } L &= \dfrac{N\Phi}{i} \\
	&= \dfrac{NBA}{i} \\
	&= \dfrac{N\left(\dfrac{\mu_0 Ni}{l}\right)A}{i} \\
	&= \mu_0 \dfrac{N^2 A}{l}
\end{flalign*}

ដូចនេះអាំងឌុចតង់នៃបូប៊ីនគឺ
\begin{center}
	$\boxed{L = \mu_0 \dfrac{N^2 A}{l}}$ \quad (១៦)
\end{center}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item $L$ ៖ អាំងឌុចតង់ $(\text{H})$
	\item $\mu_0 = 4\pi \times 10^{-7} \, \text{T}\cdot\text{m/A}$ ៖ ជម្រាបម៉ាញេទិក
	\item $N$ ៖ ចំនួនស្ពៀនៃរបុំ $(\text{ស្ពៀ ឬ ជុំ})$
	\item $A = \pi R^2 = \pi \left(\frac{d}{2}\right)^2$ ៖ ផ្ទៃនៃសូលេណូអ៊ីត $(\text{m}^2)$
	\item $l$ ៖ ប្រវែងសូលេណូអ៊ីត $(\text{m})$
\end{itemize}

\vspace{0.7em}
\noindent
\begin{formulabox}[breakable=false]{
	និយមន័យអាំងឌុចតង់ }
	\textbf{\textit{\textcolor{brownheader}{អាំងឌុចតង់}}} គឺជាលក្ខណៈធរណីមាត្រនៃរបុំខ្សែចម្លងដែលត្រូវបានប្រែប្រួលចរន្តអគ្គិសនី និងបង្កើតបានកម្លាំងអគ្គិសនីចលករអាំងឌ្វីនៅក្នុងរបុំរបស់វាផ្ទាល់ ឬរបុំណាមួយ ។
\end{formulabox}

\khmersubtopic{២.៣. កម្លាំងអគ្គិសនីចលករអូតូអាំងឌ្វី}

តាមច្បាប់ផារ៉ាដេ ៖
\begin{center}
	$e = -N\dfrac{d\Phi}{dt}$
\end{center}

យើងមាន (១៥) ៖ $N\Phi = Li$ \quad (ធ្វើដេរីវេអង្គទាំងពីរ)

នាំឱ្យ ៖
\begin{center}
	$N\dfrac{d\Phi_B}{dt} = L\dfrac{di}{dt}$
\end{center}

យើងបាន ៖
\begin{center}
	$e = -L\dfrac{di}{dt}$
\end{center}

ដូចនេះកម្លាំងអគ្គិសនីចលករអូតូអាំងឌ្វីខណៈគឺ
\begin{center}
	$\boxed{e = -L\dfrac{di}{dt}}$ \quad (១៧)
\end{center}

យើងបានកម្លាំងអគ្គិសនីចលករអូតូអាំងឌ្វីមធ្យមគឺ
\begin{center}
	$\boxed{E = -L\dfrac{\Delta I}{\Delta t}}$ \quad (១៨)
\end{center}

\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
	\item $e$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីខណៈនៃរបុំ $(\text{V})$
	\item $E$ ៖ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីមធ្យមនៃរបុំ $(\text{V})$
	\item $L$ ៖ អាំងឌុចតង់ $(\text{H})$
	\item $\frac{di}{dt}$ ៖ អត្រាចរន្តឆ្លងកាត់របុំ $(\text{A/s})$
	\item $\Delta I = I_2 - I_1$ ៖ បម្រែបម្រួលចរន្តឆ្លងកាត់របុំ $(\text{A})$
	\item $\Delta t = t_2 - t_1$ ៖ បម្រែបម្រួលរយៈពេល $(\text{s})$
\end{itemize}

\khmertopic{៣. ថាមពលម៉ាញេទិក}
ដើម្បីឱ្យមានចរន្តរត់នៅក្នុងប៊ូប៊ីន (ឬរបុំខ្សែចម្លង) យើងត្រូវផ្តល់ថាមពលអគ្គិសនីទៅឱ្យវា។ នៅពេលដែលប៊ូប៊ីនត្រូវបានឆ្លងកាត់ដោយចរន្តអគ្គិសនី វាស្តុកថាមពលមួយដែលមានឈ្មោះថា \textbf{\textit{\textcolor{brownheader}{ថាមពលម៉ាញេទិច}}}។

\vspace{0.3em}
\noindent
\textbf{ជំហានទី១ ៖ ប៉ូតង់ស្យែលរវាងគោលទាំងពីរនៃប៊ូប៊ីន}

បើចរន្តអគ្គិសនី $i$ ដែលរត់នៅក្នុងប៊ូប៊ីនកើនឡើង នោះភ្លុចម៉ាញេទិចឆ្លងកាត់ប៊ូប៊ីនក៏ប្រែប្រួលដែរ ហើយបង្កើតបានកម្លាំងអគ្គិសនីចលករអូតូអាំងឌ្វី។ ហេតុនេះមានបម្រែបម្រួលប៉ូតង់ស្យែលរវាងគោលទាំងពីរនៃប៊ូប៊ីន ៖
\begin{center}
	$\boxed{V_{ab} = L\dfrac{di}{dt}}$ \quad (១៩)
\end{center}

\vspace{0.3em}
\noindent
\textbf{ជំហានទី២ ៖ អានុភាព និងថាមពលក្នុងខណៈ $dt$}

អានុភាពអគ្គិសនីនៅខណៈណាមួយដែលប៊ូប៊ីនទទួលបាន ៖
\begin{center}
	$p = V_{ab}\,i = Li\dfrac{di}{dt}$
\end{center}

\defbox{
	\raggedright
	\textbf{Editor's Note:	} រូបមន្តថាមពលគឺ $E=Pt$ បើគេសិក្សាមួយផ្នែកតូចនោះគេបាន $dE=pdt$។
}

ដូចនេះថាមពលដែលប៊ូប៊ីនទទួលបានក្នុងខណៈ $dt$ គឺ ៖
\begin{center}
	$dE_L = p\,dt = \left( Li\dfrac{di}{dt} \right) dt = Li\,di$
\end{center}

\vspace{0.3em}
\noindent
\textbf{ជំហានទី៣ ៖ ថាមពលសរុបដែលស្តុកក្នុងប៊ូប៊ីន}

ដើម្បីគណនាថាមពលសរុប យើងធ្វើអាំងតេក្រាលកំណត់ពីតម្លៃចរន្ត $0$ ទៅតម្លៃចរន្តថេរ $I$ នៅក្នុងប៊ូប៊ីន ៖
\begin{flalign*}
	E_L &= \int_0^I dE_L = \int_0^I Li\,di \\
	&= L \left[ \dfrac{1}{2}i^2 \right]_0^I \\
	&= L \left( \dfrac{1}{2}I^2 - \dfrac{1}{2}(0)^2 \right) \\
	&= \dfrac{1}{2}LI^2
\end{flalign*}

\vspace{0.5em}
\noindent
\begin{formulabox}[breakable=false]{
	ថាមពលម៉ាញេទិចដែលស្តុកក្នុងប៊ូប៊ីន}
	\begin{center}
		$\boxed{E_L = \dfrac{1}{2}LI^2}$ \quad (២០)
	\end{center}
	\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
		\item $V_{ab}$ ៖ តង់ស្យុងរវាងគោលទាំងពីរនៃរបុំ $(\mathrm{V})$
		\item $L$ ៖ អាំងឌុចតង់ $(\mathrm{}m{H})$
		\item $\dfrac{di}{dt}$ ៖ អត្រាចរន្តឆ្លងកាត់របុំ $(\mathrm{A/s})$
		\item $E_L$ ៖ ថាមពលម៉ាញេទិច $(\mathrm{J})$
		\item $i$ ៖ ចរន្តអគ្គិសនីដែលរត់នៅក្នុងរបុំខ្សែចម្លង ឬសូលេណូអ៊ីត $(\mathrm{A})$ ។
	\end{itemize}
\end{formulabox}

\vspace{0.5em}
\defbox{
	\raggedright
	\textbf{ចំណាំ៖	} ថាមពល $E_L$ សមាមាត្រនឹងអាំងឌុចតង់ $L$ និងការេនៃចរន្ត $I^2$ ។ ថាមពលនេះមិនបាត់បង់ក្នុងរូបរាងជាកម្តៅទេ គឺវាស្តុកទុកនៅក្នុងប៊ូប៊ីន ហើយត្រូវបានបញ្ចេញមកវិញនៅពេលចរន្តថយចុះ (ក្នុងរយៈពេលខ្លី)។
}

\khmersubtopic{៤.១. ករណីបិទកុងតាក់ $S_1$}

តាមច្បាប់គៀឆុក ({\englishfont Kirchhoff's Loop Rule / Kirchhoff's Voltage Law}) ចំពោះតង់ស្យុងមួយជុំ ៖
\begin{center}
	$-\mathcal{E} + V_L + V_R = 0$
\end{center}

ដោយ ៖
\begin{flalign*}
	V_L &= L\dfrac{di}{dt} \\
	V_R &= Ri
\end{flalign*}

ជំនួសចូលក្នុងច្បាប់គៀឆុក យើងបាន ៖
\begin{flalign*}
	-\mathcal{E} + L\dfrac{di}{dt} + Ri &= 0 \\[0.4em]
	L\dfrac{di}{dt} &= \mathcal{E} - Ri
\end{flalign*}

ចែកអង្គទាំងពីរនឹង $L$ រួចដាក់ $R$ ជាកត្តារួម ៖
\begin{flalign*}
	\dfrac{di}{dt} &= \dfrac{R}{L}\left( \dfrac{\mathcal{E}}{R} - i \right)
\end{flalign*}

\vspace{0.3em}
\noindent
តាង $I_m = \dfrac{\mathcal{E}}{R}$ ជាចរន្តអតិបរមា ហើយប្រមូលអថេរ $i$ ទៅម្ខាង និង $t$ ទៅម្ខាងទៀត ៖
\begin{flalign*}
	\dfrac{di}{dt} &= \dfrac{R}{L}(I_m - i) \\[0.4em]
	\dfrac{di}{I_m - i} &= \dfrac{R}{L}\,dt \\[0.4em]
	\dfrac{di}{i - I_m} &= -\dfrac{R}{L}\,dt
\end{flalign*}

នៅខណៈ $t = 0$ ចរន្ត $i = 0$ ។ ដូចនេះធ្វើអាំងតេក្រាលកំណត់ពី $0$ ទៅ $i$ ចំពោះអង្គខាងឆ្វេង និងពី $0$ ទៅ $t$ ចំពោះអង្គខាងស្តាំ ៖
\begin{center}
	$\displaystyle\int_0^i \dfrac{di'}{i' - I_m} = -\dfrac{R}{L}\int_0^t dt'$
\end{center}

\vspace{0.3em}
\noindent
\begin{formulabox}[breakable=false]{
	រូបមន្តជំនួយសម្រាប់អាំងតេក្រាល }
	\begin{itemize}[itemsep=3pt, topsep=0pt, parsep=0pt, leftmargin=1.5em]
		\item $\displaystyle\int \dfrac{du}{u} = \ln|u| + C \implies \int_a^b \dfrac{dx}{x - c} = \Big[\ln|x - c|\Big]_a^b$
		\item $\displaystyle\int_0^t dt' = \Big[t'\Big]_0^t = t$
		\item $\ln A - \ln B = \ln\left(\dfrac{A}{B}\right)$ \quad និង \quad $\ln X = Y \iff X = e^Y$
	\end{itemize}
\end{formulabox}

\vspace{0.3em}
\noindent
គណនាអាំងតេក្រាលអង្គទាំងសងខាង រួចដោះស្រាយរក $i$ ៖
\begin{flalign*}
	\Big[ \ln|i' - I_m| \Big]_0^i &= -\dfrac{R}{L} \Big[ t' \Big]_0^t \\[0.5em]
	\ln|i - I_m| - \ln|0 - I_m| &= -\dfrac{R}{L}(t - 0) \\[0.5em]
	\ln\left( \dfrac{I_m - i}{I_m} \right) &= -\dfrac{R}{L}t \qquad (\text{ព្រោះ } i < I_m) \\[0.5em]
	\dfrac{I_m - i}{I_m} &= e^{-\frac{R}{L}t} \\[0.5em]
	1 - \dfrac{i}{I_m} &= e^{-\frac{R}{L}t} \\[0.5em]
	i &= I_m \left( 1 - e^{-\frac{R}{L}t} \right)
\end{flalign*}

តាង $\tau_{RL} = \dfrac{L}{R}$ គឺជាថេរពេលនៃសៀគ្វី $RL$។

\vspace{0.5em}
\noindent
\begin{formulabox}[breakable=false]{
	ចរន្តក្នុងសៀគ្វី $RL$ ពេលបិទកុងតាក់ }
	\begin{center}
		$\boxed{i(t) = I_m \left( 1 - e^{-\frac{t}{\tau_{RL}}} \right)}$ \quad (២១)
	\end{center}
	\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
		\item $i(t)$ ៖ ចរន្តនៅខណៈណាមួយ $(\mathrm{A})$
		\item $I_m \text{ ឬ } I_{\max} = \dfrac{E}{R}$ ៖ ចរន្តអតិបរមានៃសៀគ្វី $(\mathrm{A})$
		\item $\tau_{RL} = \dfrac{L}{R}$ ៖ ថេរពេលនៃសៀគ្វី $RL$ $(\mathrm{s})$
	\end{itemize}
\end{formulabox}

\khmersubtopic{៤.២. ករណីបើកកុងតាក់ $S_1$ (ករណីបិទកុងតាក់ $S_2$ ដោយចំហកុងតាក់ $S_1$ បន្ទាប់ពីប៊ូប៊ីនផ្ទុកពេញ)}

តាមច្បាប់គៀឆុក ({\englishfont Kirchhoff's Loop Rule / Kirchhoff's Voltage Law}) ចំពោះតង់ស្យុងមួយជុំ ៖
\begin{center}
	$V_{ab} + V_{bc} + V_{ca} = 0$
\end{center}

ដោយ ៖
\begin{flalign*}
	V_{ab} &= V_R = Ri \\
	V_{bc} &= V_L = L\dfrac{di}{dt} \\
	V_{ca} &= 0\,\mathrm{V} \quad (\text{គ្មានប្រភពអគ្គិសនីក្នុងសៀគ្វី})
\end{flalign*}

យើងបាន ៖
\begin{flalign*}
	Ri + L\dfrac{di}{dt} + 0 &= 0 \\[0.4em]
	L\dfrac{di}{dt} &= -Ri
\end{flalign*}

ចែកអង្គទាំងពីរនឹង $L$ រួចប្រមូលអថេរ $i$ ទៅម្ខាង និង $t$ ទៅម្ខាងទៀត ៖
\begin{flalign*}
	\dfrac{di}{dt} &= -\dfrac{R}{L}\,i \\[0.4em]
	\dfrac{di}{i} &= -\dfrac{R}{L}\,dt
\end{flalign*}

នៅខណៈ $t = 0$ ចរន្តមានតម្លៃ $I_m$ ។ ដូចនេះធ្វើអាំងតេក្រាលកំណត់ពី $I_m$ ទៅ $i$ ចំពោះអង្គខាងឆ្វេង និងពី $0$ ទៅ $t$ ចំពោះអង្គខាងស្តាំ ៖
\begin{center}
	$\displaystyle\int_{I_m}^i \dfrac{di'}{i'} = -\dfrac{R}{L}\int_0^t dt'$
\end{center}

\vspace{0.3em}
\noindent
\begin{formulabox}[breakable=false]{
	រូបមន្តជំនួយសម្រាប់អាំងតេក្រាល }
	\begin{itemize}[itemsep=3pt, topsep=0pt, parsep=0pt, leftmargin=1.5em]
		\item $\displaystyle\int_{I_m}^i \dfrac{di'}{i'} = \Big[\ln|i'|\Big]_{I_m}^i = \ln(i) - \ln(I_m) = \ln\left(\dfrac{i}{I_m}\right)$
		\item $\displaystyle -\dfrac{R}{L}\int_0^t dt' = -\dfrac{R}{L}\Big[t'\Big]_0^t = -\dfrac{R}{L}\,t$
		\item $\displaystyle\ln\left(\dfrac{i}{I_m}\right) = -\dfrac{R}{L}\,t \iff \dfrac{i}{I_m} = e^{-\frac{R}{L}t}$
	\end{itemize}
\end{formulabox}

\vspace{0.3em}
\noindent
គណនាអាំងតេក្រាលអង្គទាំងសងខាង រួចដោះស្រាយរក $i$ ៖
\begin{flalign*}
	\Big[ \ln|i'| \Big]_{I_m}^i &= -\dfrac{R}{L} \Big[ t' \Big]_0^t \\[0.5em]
	\ln(i) - \ln(I_m) &= -\dfrac{R}{L}(t - 0) \\[0.5em]
	\ln\left( \dfrac{i}{I_m} \right) &= -\dfrac{R}{L}\,t \\[0.5em]
	\dfrac{i}{I_m} &= e^{-\frac{R}{L}t} \\[0.5em]
	i &= I_m\,e^{-\frac{R}{L}t}
\end{flalign*}

តាង $\tau_{RL} = \dfrac{L}{R}$ គឺជាថេរពេលនៃសៀគ្វី $RL$ (គិតជា $\mathrm{s}$)។

\vspace{0.5em}
\noindent
\begin{formulabox}[breakable=false]{
	ចរន្តក្នុងសៀគ្វី $RL$ ពេលបិទកុងតាក់ $S_2$ }
	\begin{center}
		$\boxed{i(t) = I_m\,e^{-\frac{t}{\tau_{RL}}}}$ \quad (២២)
	\end{center}
	\begin{itemize}[itemsep=2pt, topsep=4pt, parsep=0pt, leftmargin=1.6em]
		\item $i(t)$ ៖ ចរន្តនៅខណៈណាមួយ $(\mathrm{A})$
		\item $I_m \text{ ឬ } I_{\max} = \dfrac{\mathcal{E}}{R}$ ៖ ចរន្តអតិបរមានៃសៀគ្វី $(\mathrm{A})$
		\item $\tau_{RL} = \dfrac{L}{R}$ ៖ ថេរពេលនៃសៀគ្វី $RL$ $(\mathrm{s})$
	\end{itemize}
\end{formulabox}

\vspace{0.6em}
\noindent
តាមច្បាប់គៀឆុក ({\englishfont Kirchhoff's Voltage Law}) ចំពោះតង់ស្យុងក្នុងកង់បិទជិត ៖
\begin{center}
	$\mathcal{E} - v_C = 0 \implies v_C = \mathcal{E}$
\end{center}

\noindent
នៅពេលកុងដង់សាទ័រសាកបន្ទុកពេញ (ស្ថានភាពនឹងនរ / {\englishfont Steady State}) ៖
\begin{itemize}[itemsep=4pt, topsep=2pt, parsep=0pt, leftmargin=1.6em]
	\item \textbf{តង់ស្យុងអតិបរមារវាងគោលកុងដង់សាទ័រ ៖}
	\begin{center}
		$V_C = \mathcal{E}$
	\end{center}
	\item \textbf{បន្ទុកអគ្គិសនីអតិបរមាដែលស្តុកក្នុងកុងដង់សាទ័រ ៖}
	\begin{center}
		$\boxed{Q_0 = C\mathcal{E}}$ \quad (២៣)
	\end{center}
	\item \textbf{ចរន្តអគ្គិសនីក្នុងសៀគ្វី ៖} នៅពេលបន្ទុកពេញ គ្មានចរន្តរត់ទៀតទេ គឺ $i = 0$ ។
	\item \textbf{ថាមពលអគ្គិសនីដែលស្តុកក្នុងដែនអគ្គិសនីនៃកុងដង់សាទ័រ ៖}
	\begin{center}
		$\boxed{E_E = \dfrac{1}{2}C\mathcal{E}^2 = \dfrac{Q_0^2}{2C}}$ \quad (២៤)
	\end{center}
	\item \textbf{ថាមពលម៉ាញេទិកក្នុងប៊ូប៊ីន ៖} ដោយសារគ្មានចរន្តឆ្លងកាត់ប៊ូប៊ីន នោះ $E_B = 0$ ។
\end{itemize}

\vspace{0.4em}
\defbox{
	\raggedright
	\textbf{\textcolor{brownheader}{លក្ខខណ្ឌដើមសម្រាប់ដំណាក់កាលបន្ទាប់ (Initial Conditions for $t = 0$) ៖}}\par\vspace{3pt}
	នៅខណៈពេលដែលគេបង្វែរកុងតាក់ $S$ ពី $a$ ទៅ $b$ (កំណត់យកជាខណៈដើម $t = 0$) ៖
	\[ q(0) = Q_0 = C\mathcal{E} \qquad \text{និង} \qquad i(0) = 0 \]
	ថាមពលសរុបនៃសៀគ្វីនៅខណៈដើមនេះ គឺស្មើនឹងថាមពលអគ្គិសនីសុទ្ធ ៖
	\[ E = E_E = \dfrac{Q_0^2}{2C} \]
}

\noindent
តាមច្បាប់គៀឆុកចំពោះកង់បិទជិតខាងស្តាំ ៖
\begin{center}
	$-v_C + v_L = 0$
\end{center}

ដោយ ៖
\begin{flalign*}
	v_C &= \dfrac{q}{C} \\
	v_L &= L\dfrac{di}{dt} = L\dfrac{d}{dt}\left(-\dfrac{dq}{dt}\right) = -L\dfrac{d^2q}{dt^2} \qquad \left(\text{ព្រោះ } i = -\dfrac{dq}{dt} \text{ កុងដង់សាទ័របន្ទោរបន្ទុក}\right)
\end{flalign*}

យើងបាន ៖
\begin{flalign*}
	-\dfrac{q}{C} - L\dfrac{d^2q}{dt^2} &= 0 \\[0.4em]
	L\dfrac{d^2q}{dt^2} + \dfrac{q}{C} &= 0
\end{flalign*}

ចែកអង្គទាំងពីរនឹង $L$ យើងទទួលបានសមីការឌីផេរ៉ង់ស្យែល ៖
\begin{center}
	$\boxed{\dfrac{d^2q}{dt^2} + \dfrac{1}{LC}q = 0}$ \quad (២៥)
\end{center}

\vspace{0.2em}
\noindent
សមីការ (២៥) មានរាងដូចសមីការឌីផេរ៉ង់ស្យែលនៃលំយោលអាម៉ូនិចក្នុងមេកានិច ៖
\begin{center}
	$\dfrac{d^2x}{dt^2} + \omega^2 x = 0 \implies x(t) = x_m\cos(\omega t + \varphi)$
\end{center}

\vspace{0.3em}
\noindent
ដូចនេះ ចម្លើយទូទៅនៃបន្ទុកអគ្គិសនីដែលស្តុកដោយកុងដង់សាទ័រគឺ ៖
\begin{center}
	$\boxed{q(t) = q_{\max}\cos(\omega t + \varphi)}$ \quad (២៦)
\end{center}

\vspace{0.2em}
\noindent
និងមានកន្សោមអាំងតង់ស៊ីតេចរន្តអគ្គិសនីឆ្លងកាត់សៀគ្វី ៖
\begin{center}
	$\boxed{i(t) = \dfrac{dq}{dt} = -\omega q_{\max}\sin(\omega t + \varphi) = -I_{\max}\sin(\omega t + \varphi)}$ \quad (២៧)
\end{center}

$\blacktriangleright$ \textbf{\textcolor{khmerred}{\large ថាមពលនៃសៀគ្វី $LC$ ({\englishfont Energy of an $LC$ Circuit}) ៖}}
\par\vspace{3pt}
ថាមពលសរុបនៃសៀគ្វី $LC$ គឺជាផលបូករវាងថាមពលម៉ាញេទិកដែលស្តុកក្នុងប៊ូប៊ីន និងថាមពលអគ្គិសនីដែលស្តុកក្នុងកុងដង់សាទ័រ ៖
\begin{center}
	$\boxed{E = E_L + E_C = \dfrac{1}{2}Li^2 + \dfrac{1}{2}Cv^2}$ \quad (២៩)
\end{center}

\vspace{0.3em}

\begin{formulabox}[breakable=false, fonttitle=\normalfont]{
    \khmermoul បរិមាណ និងខ្នាតក្នុងកន្សោមថាមពលសៀគ្វី $LC$}
	\begin{itemize}[itemsep=3pt, topsep=2pt, parsep=0pt, leftmargin=1.6em]
		\item $E$ ៖ ថាមពលអេឡិចត្រូម៉ាញេទិកសរុបនៃសៀគ្វី $(\mathrm{J})$
		\item $E_L = \dfrac{1}{2}Li^2$ ៖ ថាមពលម៉ាញេទិកនៃអាំងឌុចទ័រ (ប៊ូប៊ីន) $(\mathrm{J})$
		\item $E_C = \dfrac{1}{2}Cv^2 = \dfrac{q^2}{2C}$ ៖ ថាមពលអគ្គិសនីនៃកុងដង់សាទ័រ $(\mathrm{J})$
		\item $\tau_{LC} = T = 2\pi\sqrt{LC}$ ៖ ថេរពេល (ខួបផ្ទាល់) នៃសៀគ្វី $LC$ $(\mathrm{s})$
		\item $L$ ៖ អាំងឌុចតង់នៃប៊ូប៊ីន $(\mathrm{H})$
		\item $C$ ៖ កាប៉ាស៊ីតេនៃកុងដង់សាទ័រ $(\mathrm{F})$
		\item $i$ ៖ អាំងតង់ស៊ីតេចរន្តនៅខណៈណាមួយ $(\mathrm{A})$
		\item $v \text{ ឬ } V$ ៖ តង់ស្យុងរវាងគោលទាំងពីរនៃកុងដង់សាទ័រ $(\mathrm{V})$
	\end{itemize}
\end{formulabox}

\vspace{0.4em}
\defbox{
	\raggedright
	\textbf{\textcolor{brownheader}{ការរក្សាថាមពលក្នុងសៀគ្វី $LC$ អ៊ីដេអាល់ (Conservation of Energy) ៖}}\par\vspace{3pt}
	ក្នុងសៀគ្វី $LC$ អុីដេអាល់ (គ្មានរេស៊ីស្តង់ $R = 0$) គ្មានការបាត់បង់ថាមពលជាកម្ដៅឡើយ។ ថាមពលសរុបនៃសៀគ្វីរក្សាតម្លៃថេរជានិច្ច ៖
	\[ E = E_L + E_C = \dfrac{1}{2}Li^2 + \dfrac{q^2}{2C} = \dfrac{1}{2}LI_{\max}^2 = \dfrac{Q_0^2}{2C} = \text{ថេរ} \]
	\begin{itemize}[itemsep=2pt, topsep=1pt, parsep=0pt, leftmargin=1.4em]
		\item នៅពេល $q = \pm Q_0$ នោះ $i = 0$ ៖ ថាមពលទាំងអស់ជាថាមពលអគ្គិសនី $E = E_{C,\max} = \dfrac{Q_0^2}{2C}$
		\item នៅពេល $q = 0$ នោះ $i = \pm I_{\max}$ ៖ ថាមពលទាំងអស់ជាថាមពលម៉ាញេទិក $E = E_{L,\max} = \dfrac{1}{2}LI_{\max}^2$
	\end{itemize}
}

\clearpage
\khmerlesson{មេរៀនទី៤៖ ចរន្តឆ្លាស់ (\englishfont Alternating Current)}
\tanbgdivider

ឧបករណ៍អគ្គិសនីអាចដំណើរការបានលុះត្រាតែយើងភ្ជាប់វាទៅនឹងប្រភពនៃថាមពលអគ្គិសនី។ ការប្រើប្រាស់ប្រភពអគ្គិសនីនេះត្រូវបានចែកជាពីរប្រភេទគឺ ប្រភពអគ្គិសនីចរន្តជាប់ (ចរន្តជាប់ គឺជាចរន្តដែលមានតម្លៃថេរ ហើយផ្លាស់ទីក្នុងទិសដៅមួយច្បាស់លាស់) និងប្រភពអគ្គិសនីចរន្តឆ្លាស់ (ចរន្តឆ្លាស់ គឺជាចរន្តដែលមានតម្លៃប្រែប្រួលតាមពេល ហើយផ្លាស់ទីដោយប្ដូរទិសដៅពីរដងក្នុងមួយខួប)។

\vspace{0.3em}
នៅក្នុងឆ្នាំ ១៨៨០ មានវិស្វករអគ្គិសនីពីររូបនៅក្នុងសហរដ្ឋអាមេរិក គឺលោក ថូម៉ាស់ អេឌីសុន ({\englishfont Thomas Edison}) និងលោក ចច វ៉េសទីងហោស៍ ({\englishfont George Westinghouse}) បានពិភាក្សាគ្នារកវិធីដ៏ល្អបំផុតចំពោះការផលិតថាមពលអគ្គិសនី និងការផ្គត់ផ្គង់ថាមពលអគ្គិសនីនៃប្រភពអគ្គិសនីទាំងពីរប្រភេទនេះ។ បន្ទាប់ពីធ្វើការពិភាក្សាដេញដោលគ្នារួចមក ប្រភពអគ្គិសនីចរន្តឆ្លាស់ គឺជាជម្រើសដ៏ល្អបំផុតនៅក្នុងការផ្គត់ផ្គង់ទៅដល់អ្នកប្រើប្រាស់ ពីព្រោះតង់ស្យុង និងចរន្តដែលផលិតបានគឺមានតម្លៃប្រែប្រួលតាមពេល មានទម្រង់ជាអនុគមន៍ស៊ីនុយសូអុីត។ លោក ចច វ៉េសទីងហោស៍ បានលើកឡើងថា យើងអាចប្រើប្រាស់ត្រង់ស្វូជាមួយនឹងប្រភពអគ្គិសនីចរន្តឆ្លាស់ ដែលជាឧបករណ៍សម្រាប់ទម្លាក់ ឬដំឡើងតង់ស្យុង (ដែលនឹងសិក្សានៅក្នុងមេរៀននេះ) ដែលមិនអាចប្រើជាមួយចរន្តជាប់បានទេ គឺអាចធ្វើឱ្យយើងអាចដឹកជញ្ជូនថាមពលអគ្គិសនីនៅចម្ងាយឆ្ងាយ (ពីកន្លែងផលិត ទៅកន្លែងប្រើប្រាស់)។\\

\khmertopic{១. ការផលិតកម្លាំងអគ្គិសនីចរន្តឆ្លាស់ស៊ីនុយសូអុីត}

\noindent
ស៊ុមខ្សែចម្លងមួយមានផ្ទៃ ($\vec{A}$) និងរុំដោយរបុំខ្សែចម្លងមាន $N$ ស្ពៀ ត្រូវបានដាក់នៅក្នុងដែនម៉ាញេទិកឯកសណ្ឋាន ($\vec{B}$) និងវិលដោយប្រេកង់មុំ ($\omega = 2\pi f$) ដូចបង្ហាញក្នុងរូបទី៤២ ៖

\vspace{0.4em}
\begin{center}
	\khmerphoto[0.72\linewidth]{img/motor2.0.png}{រូបទី៤២៖ ស៊ុមវិលនៅក្នុងដែនម៉ាញេទិក}
\end{center}

\vspace{0.4em}
\noindent
តាមច្បាប់ផារ៉ាដេ កម្លាំងអគ្គិសនីចលករអាំងឌ្វីដែលផលិតដោយស៊ុមវិលគឺ ៖
\begin{equation*}
	e = -N\dfrac{d\Phi_B}{dt}
\end{equation*}

\noindent
ដោយ ៖ $\Phi_B = BA\cos(\theta) = BA\cos(\omega t)$ \quad (ព្រោះ $\theta = \omega t$)

\vspace{0.3em}
\noindent
យើងបាន ៖
\begin{align*}
	e &= -N\dfrac{d}{dt}\Big(BA\cos(\omega t)\Big) \\[0.4em]
	&= -NBA\Big(-\omega\sin(\omega t)\Big) \\[0.4em]
	&= NBA\omega\sin(\omega t)
\end{align*}

\vspace{0.2em}
\noindent
ដូចនេះ កម្លាំងអគ្គិសនីចលករឆ្លាស់ស៊ីនុយសូអុីតដែលផលិតដោយស៊ុមវិលគឺ ៖
\begin{center}
	$\boxed{e = NBA\omega\sin(\omega t) \quad \text{ឬ} \quad e = e_{\max}\sin(\omega t)}$ \quad (៣០)
\end{center}

\vspace{0.3em}
\noindent
\begin{minipage}[t]{0.48\textwidth}
	\begin{itemize}[itemsep=2pt, topsep=1pt, parsep=0pt, leftmargin=1.5em]
		\item $e$ ៖ កម្លាំងអគ្គិសនីចលករខណៈ $(\mathrm{V})$
		\item $N$ ៖ ចំនួនរបុំនៃស៊ុម (ជុំ)
		\item $B$ ៖ ដែនម៉ាញេទិកឯកសណ្ឋាន $(\mathrm{T})$
		\item $A$ ៖ ផ្ទៃនៃស៊ុមខ្សែចម្លង $(\mathrm{m}^2)$
	\end{itemize}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.48\textwidth}
	\begin{itemize}[itemsep=2pt, topsep=1pt, parsep=0pt, leftmargin=1.5em]
		\item $\omega = 2\pi f = \dfrac{2\pi}{T}$ ៖ ប្រេកង់មុំ ឬពុលសាសុង $(\mathrm{rad/s})$
		\item $e_{\max} = NBA\omega$ ៖ តម្លៃអតិបរមានៃកម្លាំងអគ្គិសនីចលករ $(\mathrm{V})$
	\end{itemize}
\end{minipage}

	\begin{center}
		$\boxed{v = e = V_{\max}\sin(\omega t)}$ \quad (៣១)
	\end{center}
	\begin{itemize}[itemsep=2pt, topsep=1pt, parsep=0pt, leftmargin=1.5em]

	\end{itemize}
}

\noindent
\textbf{សំទួយចរន្តមធ្យម} ({\englishfont The rectifier average current}) ត្រូវបានកំណត់សម្រាប់ពេញរលក គឺមានតម្លៃស្មើនឹងអត្រាបរិមាណបន្ទុកសរុបដែលរត់ឆ្លងកាត់កាល់វ៉ាណូម៉ែត្រក្នុងរយៈពេលណាមួយ (ផ្ទៃនៅក្នុងរូបទី៤៦.ខ) ។ សំទួយចរន្តមធ្យមនេះត្រូវបានតាងដោយ $I_{\mathrm{rav}}$ មានតម្លៃតិចជាងតម្លៃចរន្តអតិបរមា $I_{\max}$ និងមានរូបមន្ត ៖
\begin{center}
	$\boxed{I_{\mathrm{rav}} = \frac{2}{\pi}I_{\max}}$ \quad (៣២)
\end{center}

		\item $\dfrac{2}{\pi}$ ៖ តម្លៃជាមធ្យមនៃ $|\sin(\omega t)|$ ឬ $|\cos(\omega t)|$

	\end{itemize}

}

\khmersubtopic{១.៣. តម្លៃប្រសិទ្ធ (\englishfont Root-Mean-Square Value)}

ដើម្បីសិក្សាពីរលកស៊ីនុយសូអុីតនៃចរន្ត ឬតង់ស្យុងនៃចរន្តឆ្លាស់ ដែលប្រែប្រួលតាមពេលដែលអាចមានតម្លៃវិជ្ជមាន ឬអវិជ្ជមាន គឺត្រូវបានកំណត់ដោយ\textbf{តម្លៃប្រសិទ្ធ}។ ដើម្បីរកចរន្តប្រសិទ្ធ យើងត្រូវ ៖
