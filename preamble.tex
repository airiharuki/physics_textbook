\newcommand{\khmerdigit}[1]{%
  \ifcase#1
    ០%
  \or ១%
  \or ២%
  \or ៣%
  \or ៤%
  \or ៥%
  \or ៦%

  \fi
}

% NOTE 1: e-TeX's \numexpr "/" ROUNDS to the nearest integer rather than
% truncating (so \numexpr 15/10\relax = 2, not 1), which silently corrupted
% every multi-digit conversion here (page numbers, section numbers, etc.
% for any value >= 10 whose tens digit rounds up). \count register
% \divide truncates toward zero like normal integer division, so we use
% that instead.
%
% NOTE 2: the argument is normalized into a count register *before*
% anything else happens. If an unexpanded macro like \arabic{page} or
% \value{page} is used directly inside \ifnum#1<10 and again inside a
% later \count assignment, the two re-expansions of #1 can desync (the
% \ifnum comparison and the register copy end up seeing different partial
% expansions), silently truncating the result to one digit. Assigning
% #1 straight into a register forces one clean, complete expansion that
% every later step then reads back from the register instead of re-
% expanding #1 itself.
\newcommand{\khmer@donums}[2]{\khmernumraw{#1}\khmerdigit{#2}}
\newcommand{\khmernumraw}[1]{%
  \count9=#1\relax
  \ifnum\count9<10 %
    \khmerdigit{\the\count9}%
  \else
    \begingroup
      \count8=\count9
      \divide\count8 by 10 %
      \count7=\count8
      \multiply\count7 by 10 %
      \advance\count9 by -\count7 %
      % capture quotient+remainder as literal text (not a shared macro
      % name) before recursing, so nested calls can't clobber this level
      \xdef\khmer@qr{{\the\count8}{\the\count9}}%
    \endgroup
    \expandafter\khmer@donums\khmer@qr
  \fi
}

\makeatother

\newcommand{\khmernum}[1]{%
  {\khmerpagenum ទំព័រទី \khmernumraw{#1}}%
}
